```python
"""SSE Event Models."""

from typing import Any, Dict, List, Optional
from pydantic import BaseModel, Field


class PlayerDelta(BaseModel):
    """Delta for a single player."""
    player_id: str
    rank: int
    score: float
    moves: int


class SnapshotEvent(BaseModel):
    """Full board state snapshot."""
    type: str = "snapshot"
    board_id: str
    players: List[PlayerDelta]
    timestamp: float = Field(default_factory=lambda: __import__('time').time())


class DeltaEvent(BaseModel):
    """Delta update for one or more players."""
    type: str = "delta"
    board_id: str
    deltas: List[PlayerDelta]
    timestamp: float = Field(default_factory=lambda: __import__('time').time())


class HeartbeatEvent(BaseModel):
    """Periodic heartbeat to keep connection alive."""
    type: str = "heartbeat"
    board_id: str
    timestamp: float = Field(default_factory=lambda: __import__('time').time())


def format_sse(data: BaseModel) -> str:
    """Format a Pydantic model as an SSE event."""
    data_dict = data.dict(exclude_none=True)
    # Ensure 'type' is included if it's not in the dict (though it should be)
    if 'type' not in data_dict:
        data_dict['type'] = data.type
    
    # Serialize to JSON string
    import json
    payload = json.dumps(data_dict)
    
    return f"data: {payload}\n\n"
```

```python
"""SSE Hub for managing board event streams."""

import asyncio
import logging
from typing import Dict, Optional, Set

from apps.api.models.sse import (
    DeltaEvent,
    HeartbeatEvent,
    PlayerDelta,
    SnapshotEvent,
    format_sse,
)

logger = logging.getLogger(__name__)


class SSEHub:
    """Manages SSE connections per board."""

    def __init__(self, heartbeat_interval: float = 15.0):
        self.heartbeat_interval = heartbeat_interval
        # board_id -> set of subscriber_ids
        self._subscribers: Dict[str, Set[str]] = {}
        # subscriber_id -> asyncio.Queue
        self._queues: Dict[str, asyncio.Queue] = {}
        # subscriber_id -> asyncio.Task
        self._tasks: Dict[str, asyncio.Task] = {}
        self._running = False

    async def start(self):
        """Start the hub."""
        self._running = True
        logger.info("SSE Hub started")

    async def stop(self):
        """Stop the hub and cancel all tasks."""
        self._running = False
        for task in self._tasks.values():
            task.cancel()
        await asyncio.gather(*self._tasks.values(), return_exceptions=True)
        self._tasks.clear()
        self._queues.clear()
        self._subscribers.clear()
        logger.info("SSE Hub stopped")

    def add_subscriber(self, board_id: str, subscriber_id: str) -> asyncio.Queue:
        """Register a new subscriber for a board."""
        if board_id not in self._subscribers:
            self._subscribers[board_id] = set()
        
        self._subscribers[board_id].add(subscriber_id)
        queue: asyncio.Queue = asyncio.Queue()
        self._queues[subscriber_id] = queue
        
        # Start the background task for this subscriber
        task = asyncio.create_task(self._stream_events(board_id, subscriber_id))
        self._tasks[subscriber_id] = task
        
        logger.debug(f"Subscriber {subscriber_id} added to board {board_id}")
        return queue

    def remove_subscriber(self, board_id: str, subscriber_id: str):
        """Remove a subscriber from a board."""
        if board_id in self._subscribers:
            self._subscribers[board_id].discard(subscriber_id)
            if not self._subscribers[board_id]:
                del self._subscribers[board_id]
        
        if subscriber_id in self._queues:
            # Put a sentinel to break the stream loop
            self._queues[subscriber_id].put(None)
            del self._queues[subscriber_id]
        
        if subscriber_id in self._tasks:
            self._tasks[subscriber_id].cancel()
            del self._tasks[subscriber_id]
            
        logger.debug(f"Subscriber {subscriber_id} removed from board {board_id}")

    def push_delta(self, board_id: str, deltas: list[PlayerDelta]):
        """Push a delta event to all subscribers of a board."""
        if board_id not in self._subscribers:
            return
        
        delta_event = DeltaEvent(
            board_id=board_id,
            deltas=deltas
        )
        
        for subscriber_id in self._subscribers[board_id]:
            if subscriber_id in self._queues:
                try:
                    self._queues[subscriber_id].put_nowait(delta_event)
                except asyncio.QueueFull:
                    logger.warning(f"Queue full for subscriber {subscriber_id}, dropping delta")

    def push_snapshot(self, board_id: str, players: list[PlayerDelta]):
        """Push a snapshot event to all subscribers of a board."""
        if board_id not in self._subscribers:
            return
        
        snapshot_event = SnapshotEvent(
            board_id=board_id,
            players=players
        )
        
        for subscriber_id in self._subscribers[board_id]:
            if subscriber_id in self._queues:
                try:
                    self._queues[subscriber_id].put_nowait(snapshot_event)
                except asyncio.QueueFull:
                    logger.warning(f"Queue full for subscriber {subscriber_id}, dropping snapshot")

    async def _stream_events(self, board_id: str, subscriber_id: str):
        """Background task to stream events to a subscriber."""
        queue = self._queues.get(subscriber_id)
        if not queue:
            return
        
        try:
            last_heartbeat = asyncio.get_event_loop().time()
            
            while self._running:
                try:
                    # Wait for an event or timeout for heartbeat
                    event = await asyncio.wait_for(
                        queue.get(),
                        timeout=self.heartbeat_interval
                    )
                    
                    if event is None:
                        # Sentinel received, exit
                        break
                    
                    yield_event = format_sse(event)
                    # In a real app, you'd send this to the client via the response stream
                    # Here we just store it in a temporary list for testing purposes
                    # But since this is a background task, we can't easily yield to the endpoint.
                    # Instead, the endpoint will read from the queue.
                    # This method is just to handle heartbeats and queue draining.
                    # Actually, the endpoint reads from the queue. This task handles heartbeats.
                    
                except asyncio.TimeoutError:
                    # Send heartbeat
                    heartbeat_event = HeartbeatEvent(board_id=board_id)
                    try:
                        queue.put_nowait(heartbeat_event)
                    except asyncio.QueueFull:
                        pass
                
        except asyncio.CancelledError:
            pass
        finally:
            # Cleanup
            self.remove_subscriber(board_id, subscriber_id)


# Singleton instance
sse_hub = SSEHub()
```

```python
"""SSE Endpoint for Board Events."""

import uuid
from typing import AsyncGenerator

from fastapi import APIRouter, Query, Request
from fastapi.responses import StreamingResponse

from apps.api.services.sse_hub import sse_hub
from apps.api.models.sse import format_sse, SnapshotEvent, PlayerDelta

router = APIRouter()


@router.get("/sse")
async def sse_endpoint(
    board_id: str = Query(..., description="Board ID to subscribe to"),
    request: Request = None
):
    """
    SSE endpoint for real-time board updates.
    
    Connects to the SSE hub and streams snapshot, delta, and heartbeat events.
    """
    subscriber_id = str(uuid.uuid4())
    
    # Get the queue for this subscriber
    queue = sse_hub.add_subscriber(board_id, subscriber_id)
    
    async def event_stream() -> AsyncGenerator[str, None]:
        try:
            # Send initial snapshot (if available)
            # For now, we assume the hub will push the initial snapshot when the subscriber is added.
            # If not, we might need to fetch it from the board service.
            # Let's assume the hub handles pushing the initial snapshot.
            
            while True:
                if await request.is_disconnected():
                    break
                
                event = await queue.get()
                
                if event is None:
                    break
                
                yield format_sse(event)
                
        finally:
            # Cleanup on disconnect
            sse_hub.remove_subscriber(board_id, subscriber_id)

    return StreamingResponse(
        event_stream(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",
        }
    )
```

```python
@pytest.fixture(autouse=True)
def reset_sse_hub():
    """Reset the SSE hub before each test."""
    # Stop any running tasks
    if sse_hub._running:
```
