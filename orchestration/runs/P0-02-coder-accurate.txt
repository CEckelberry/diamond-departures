### 1. Concise Plan

1.  **Schema Design (`001_init.sql`)**: Create tables for `teams`, `players`, `games`, and `stats` based on standard baseball analytics architecture. Use `numeric(8,3)` for all statistical values. Add necessary indexes (e.g., on foreign keys and date columns) and constraints (unique team slugs).
2.  **Seed Data (`002_seed_teams.sql`)**: Insert the 30 MLB teams into the `teams` table. Use standard abbreviations and city names.
3.  **Park Factors (`003_park_factors.sql`)**: Create a table for park factors and insert a seed dataset for the 2026 season (simulated data).
4.  **JSON Assets**: Generate `packages/content/teams.json` and `packages/content/park-factors-2026.json` to serve as source-of-truth or API fixtures.
5.  **Makefile**: Add a `migrate` target that utilizes `migrate` CLI tool (standard convention) with `$DATABASE_URL`.

### 2. Exact File Edits

#### `apps/api/migrations/001_init.sql`

```sql
-- +migrate Up
-- +migrate StatementBegin

CREATE TABLE IF NOT EXISTS teams (
    id SERIAL PRIMARY KEY,
    slug VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(100) NOT NULL,
    city VARCHAR(50) NOT NULL,
    league VARCHAR(10) CHECK (league IN ('AL', 'NL')),
    division VARCHAR(20),
    established_year INTEGER,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS players (
    id SERIAL PRIMARY KEY,
    team_id INTEGER REFERENCES teams(id) ON DELETE SET NULL,
    mlb_id INTEGER UNIQUE,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    position VARCHAR(10),
    bats VARCHAR(5) CHECK (bats IN ('L', 'R', 'S')),
    throws VARCHAR(5) CHECK (throws IN ('L', 'R', 'S')),
    birth_date DATE,
    height_feet INTEGER,
    height_inches INTEGER,
    weight_lbs INTEGER,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS games (
    id SERIAL PRIMARY KEY,
    game_id INTEGER UNIQUE, -- External ID (e.g., from MLB AM)
    date DATE NOT NULL,
    home_team_id INTEGER REFERENCES teams(id),
    away_team_id INTEGER REFERENCES teams(id),
    venue_name VARCHAR(100),
    start_time TIMESTAMPTZ,
    status VARCHAR(20), -- e.g., 'FINAL', 'SCHEDULED'
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS stats (
    id SERIAL PRIMARY KEY,
    game_id INTEGER REFERENCES games(id) ON DELETE CASCADE,
    player_id INTEGER REFERENCES players(id) ON DELETE SET NULL,
    team_id INTEGER REFERENCES teams(id),
    stat_type VARCHAR(50) NOT NULL, -- e.g., 'batting', 'pitching'
    at_bats NUMERIC(8,3),
    runs NUMERIC(8,3),
    hits NUMERIC(8,3),
    doubles NUMERIC(8,3),
    triples NUMERIC(8,3),
    home_runs NUMERIC(8,3),
    rbi NUMERIC(8,3),
    stolen_bases NUMERIC(8,3),
    walks NUMERIC(8,3),
    strikeouts NUMERIC(8,3),
    earned_runs NUMERIC(8,3),
    innings_pitched NUMERIC(8,3),
    hits_allowed NUMERIC(8,3),
    walks_allowed NUMERIC(8,3),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Indexes for performance
CREATE INDEX IF NOT EXISTS idx_players_team_id ON players(team_id);
CREATE INDEX IF NOT EXISTS idx_games_date ON games(date);
CREATE INDEX IF NOT EXISTS idx_games_home_team ON games(home_team_id);
CREATE INDEX IF NOT EXISTS idx_games_away_team ON games(away_team_id);
CREATE INDEX IF NOT EXISTS idx_stats_game_id ON stats(game_id);
CREATE INDEX IF NOT EXISTS idx_stats_player_id ON stats(player_id);
CREATE INDEX IF NOT EXISTS idx_stats_team_id ON stats(team_id);

-- +migrate StatementEnd

-- +migrate Down
-- +migrate StatementBegin

DROP TABLE IF EXISTS stats;
DROP TABLE IF EXISTS games;
DROP TABLE IF EXISTS players;
DROP TABLE IF EXISTS teams;

-- +migrate StatementEnd
```

#### `apps/api/migrations/003_park_factors.sql`

```sql
-- +migrate Up
-- +migrate StatementBegin

CREATE TABLE IF NOT EXISTS park_factors (
    id SERIAL PRIMARY KEY,
    team_id INTEGER REFERENCES teams(id),
    season_year INTEGER NOT NULL,
    park_name VARCHAR(100) NOT NULL,
    factor_offense NUMERIC(8,3), -- e.g., 1.05 means 5% more offense than league avg
    factor_pitching NUMERIC(8,3),
    factor_humidity NUMERIC(8,3),
    factor_altitude NUMERIC(8,3),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(team_id, season_year)
);

-- Seed 2026 Park Factors (Simulated Data)
INSERT INTO park_factors (team_id, season_year, park_name, factor_offense, factor_pitching, factor_humidity, factor_altitude)
SELECT t.id, 2026, t.city || ' Stadium', 
       CASE 
           WHEN t.slug IN ('COL') THEN 1.15 -- Coors Field is huge hitter friendly
           WHEN t.slug IN ('LAD', 'SF', 'SD') THEN 0.95 -- Pitcher friendly coastal
           ELSE 1.00 -- League average
       END,
       CASE 
           WHEN t.slug IN ('COL') THEN 0.85
           WHEN t.slug IN ('LAD', 'SF', 'SD') THEN 1.05
           ELSE 1.00
       END,
       CASE 
           WHEN t.slug IN ('MIA', 'HOU', 'TEX', 'ATL') THEN 1.10 -- Humid
           ELSE 0.90
       END,
       CASE 
           WHEN t.slug IN ('COL') THEN 1.20 -- High altitude
           ELSE 1.00
       END
FROM teams t
ON CONFLICT (team_id, season_year) DO NOTHING;

-- +migrate StatementEnd

-- +migrate Down
-- +migrate StatementBegin

DROP TABLE IF EXISTS park_factors;

-- +migrate StatementEnd
```
