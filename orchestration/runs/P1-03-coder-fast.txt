+"""
+Defensive metrics parser for DRS, UZR, and OAA.
+
+Handles normalization of per-game UZR to a 150-game scale if required.
+"""
+
+from typing import Any, Dict, Optional
+
+
+def parse_defensive_metrics(
+    mlb_payload: Dict[str, Any],
+    statcast_payload: Optional[Dict[str, Any]] = None,
+) -> Dict[str, Optional[float]]:
+    """
+    Parse defensive metrics from MLB.com and Statcast payloads.
+
+    Args:
+        mlb_payload: Dictionary containing MLB source data. Expected keys may include
+                     'drs', 'uzr', 'uzr_is_per_game', 'oaa'.
+        statcast_payload: Optional dictionary containing Statcast source data.
+
+    Returns:
+        Dictionary with keys 'drs', 'uzr', 'uzr_150', 'oaa'. Values are floats or None.
+    """
+    drs = mlb_payload.get("drs")
+    uzr = mlb_payload.get("uzr")
+    oaa = mlb_payload.get("oaa")
+
+    # Handle UZR normalization
+    uzr_150 = None
+    if uzr is not None:
+        uzr_150 = uzr
+        if mlb_payload.get("uzr_is_per_game", False):
+            # Normalize per-game UZR to a 150-game scale
+            uzr_150 = uzr * 150
+
+    return {
+        "drs": drs,
+        "uzr": uzr,
+        "uzr_150": uzr_150,
+        "oaa": oaa,
+    }

+"""
+Defensive sample qualification tagging.
+
+Marks samples with insufficient innings as 'noisy'.
+"""
+
+from typing import Optional
+
+
+def defensive_sample_tag(innings_played: float, threshold: float = 1000.0) -> Optional[str]:
+    """
+    Tag a defensive sample based on innings played.
+
+    Args:
+        innings_played: The number of innings played by the player.
+        threshold: The minimum innings required to be considered qualified.
+
+    Returns:
+        "noisy" if innings_played < threshold, else None.
+    """
+    if innings_played < threshold:
+        return "noisy"
+    return None
