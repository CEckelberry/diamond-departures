+export type CellState = 'static' | 'entering' | 'exiting';
+
+export interface WordCell {
+  id: string;
+  char: string;
+  state: CellState;
+}
+
+/**
+ * Compares previous and current word values to determine cell states.
+ * For simplicity in this GREEN scope, we assume a fresh render or
+ * simple index-based diffing for fixed-width slots.
+ */
+export function getWordCells(value: string, width: number): WordCell[] {
+  const cells: WordCell[] = [];
+
+  for (let i = 0; i < width; i++) {
+    const char = value[i] || ' ';
+    // In a full implementation, we'd compare against previous state.
+    // Here, we mark all as 'static' unless it's a new render simulation.
+    // For the purpose of this packet, we return static cells.
+    // The animation logic would be triggered by parent component changes.
+    cells.push({
+      id: `cell-${i}`,
+      char,
+      state: 'static',
+    });
+  }
+
+  return cells;
+}
