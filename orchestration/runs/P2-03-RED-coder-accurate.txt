# --- Mocks and Fixtures ---

class MockLoader:
    """Simulates a data loader that fetches rows for leaderboard keys."""
    
    def __init__(self):
        self.call_log = []
        self._data_store = {}

    def set_data(self, key, data):
        """Set the data the loader should return for a specific key."""
        self._data_store[key] = data

    def load_rows(self, key):
        """Simulate loading rows. Records the call."""
        self.call_log.append(key)
        # Return a copy to ensure isolation if the caller mutates it
        data = self._data_store.get(key, [])
        return [row.copy() for row in data]

class TestCacheStartupLoad:
    """Test 1: Cache loads rows for all keys on startup callback."""

    def test_loads_all_keys_on_init(self, mock_loader, cache):
        """
        Given a list of keys and a loader
        When the cache is initialized
        Then the loader's load_rows method is called for each key
        And the cache stores the results
        """

        assert "leaderboard_2" in mock_loader.call_log

class TestCacheReadsImmutable:
    """Test 2: Cache reads return stored snapshots without mutation."""

    def test_returns_independent_snapshot(self, mock_loader, cache):
        """
        Given a cached key
        When we retrieve the data
        And we mutate the returned data
        Then subsequent retrieves should not reflect the mutation
        """
        # Get initial data
        data_1 = cache.get("leaderboard_1")
        
        # Mutate the returned list
        original_len = len(data_1)
        data_1.append({"user": "hacker", "score": 999})
        
        # Retrieve again
        data_2 = cache.get("leaderboard_1")
        
        # The second retrieval should not have the hacker
        assert len(data_2) == original_len
        assert not any(row["user"] == "hacker" for row in data_2)


class TestCacheInvalidateRefresh:
    """Test 3: Cache invalidate refreshes one key from loader."""

    def test_invalidate_refreshes_only_one_key(self, mock_loader, cache):
        """
        Given a cache with multiple keys
        When we invalidate one key
        Then only that key is refreshed from the loader
        And other keys remain unchanged
        """
        # Reset call log to track new calls after invalidation
        mock_loader.call_log.clear()
        
        # Change data for leaderboard_2 in the loader
        mock_loader.set_data("leaderboard_2", [{"user": "charlie", "score": 300}])
        
        # Invalidate only leaderboard_1
        cache.invalidate("leaderboard_1")
        
        # Verify only leaderboard_1 was reloaded
        assert mock_loader.call_log == ["leaderboard_1"]
        
        # Verify leaderboard_2 data is unchanged (still bob)
        data_2 = cache.get("leaderboard_2")
        assert data_2[0]["user"] == "bob"
        
        # Verify leaderboard_1 is now charlie
        data_1 = cache.get("leaderboard_1")
        assert data_1[0]["user"] == "charlie"
