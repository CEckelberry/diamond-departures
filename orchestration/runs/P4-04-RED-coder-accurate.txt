describe('Board Structure', () => {
  it('renders consistent column headers', () => {
    const { container } = render(Board, {
      props: {
        rows: Array.from({ length: 100 }, (_, i) => ({
          rank: i + 1,
          player: `Player ${i + 1}`,
          team: `Team ${i + 1}`,
          position: 'Forward',
          stat: 100 - i
        }))
      }
    });

    // Assert header exists and has expected columns
    const headers = container.querySelectorAll('th');
    expect(headers.length).toBeGreaterThanOrEqual(5);
    
    const headerTexts = Array.from(headers).map(h => h.textContent.trim().toLowerCase());
    expect(headerTexts).toContain('rank');
    expect(headerTexts).toContain('player');
    expect(headerTexts).toContain('team');
    expect(headerTexts).toContain('position');
    expect(headerTexts).toContain('stat');
  });

  it('renders 100-row list region', () => {
    const rowsData = Array.from({ length: 100 }, (_, i) => ({
      rank: i + 1,
      player: `Player ${i + 1}`,
      team: `Team ${i + 1}`,
      position: 'Forward',
      stat: 100 - i
    }));

    const { container } = render(Board, { props: { rows: rowsData } });

    // Assert list region exists
    const listRegion = container.querySelector('[data-testid="board-list"]') || 
                       container.querySelector('.board-list') ||
                       container.querySelector('tbody');
    
    expect(listRegion).toBeTruthy();

    // Assert 100 rows are rendered
    const rowElements = listRegion.querySelectorAll('tr');
    expect(rowElements.length).toBe(100);
  });
});
