package com.mlb.stats.scanner;

import com.mlb.stats.provider.Provider;
import com.mlb.stats.provider.ProviderException;
import com.mlb.stats.model.GameState;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import java.util.concurrent.Executors;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.ConcurrentHashMap;

/**
 * Scanner that detects game state changes and triggers updates.
 */
public class GameChangeScanner {
    
    private static final Logger logger = LoggerFactory.getLogger(GameChangeScanner.class);
    private static final long SCAN_INTERVAL_SECONDS = 30;
    
    private final ScheduledExecutorService executor;
    private final Provider<GameState> provider;
    private final ConcurrentHashMap<String, GameState> gameStates = new ConcurrentHashMap<>();
    
    public GameChangeScanner(Provider<GameState> provider) {
        this.provider = provider;
        this.executor = Executors.newScheduledThreadPool(2);
    }
    
    /**
     * Start monitoring games with the given game IDs.
     */
    public void startMonitoring(String... gameIds) {
        logger.info("Starting game change scanner for games: {}", String.join(", ", gameIds));
        
        for (String gameId : gameIds) {
            monitorGame(gameId);
        }
    }
    
    private void monitorGame(String gameId) {
        executor.scheduleAtFixedRate(() -> {
            try {
                GameState currentState = provider.fetch(gameId)
                    .orElseThrow(() -> new ProviderException("Game not found: " + gameId));
                
                GameState previousState = gameStates.put(gameId, currentState);
                
                if (hasGameChanged(previousState, currentState)) {
                    logger.info("Game change detected for game: {}", gameId);
                    notifyGameChangeListeners(gameId, currentState, previousState);
                }
                
            } catch (ProviderException e) {
                logger.warn("Failed to scan game: {}", gameId, e);
            }
        }, 0, SCAN_INTERVAL_SECONDS, TimeUnit.SECONDS);
    }
    
    private boolean hasGameChanged(GameState previous, GameState current) {
        if (previous == null) return true;
        
        return !previous.equals(current);
    }
    
    private void notifyGameChangeListeners(String gameId, GameState current, GameState previous) {
        // In production, this would use a listener pattern
        System.out.printf("GAME CHANGE: Game %s - Inning: %s %d, Score: %d-%d, Outs: %d%n",
            gameId,
            current.getInningState(),
            current.getInning(),
            current.getHomeScore(),
            current.getAwayScore(),
            current.getOuts());
    }
    
    /**
     * Stop all monitoring and shutdown executor.
     */
    public void stop() {
        logger.info("Stopping game change scanner");
        executor.shutdown();
        try {
            if (!executor.awaitTermination(5, TimeUnit.SECONDS)) {
                executor.shutdownNow();
            }
        } catch (InterruptedException e) {
            executor.shutdownNow();
            Thread.currentThread().interrupt();
        }
    }
}

package com.mlb.stats.model;

import java.util.List;

/**
 * Represents the current state of a baseball game.
 */
public class GameState {
    private String gameId;
    private int inning;
    private String inningState; // TOP or BOTTOM
    private int homeScore;
    private int awayScore;
    private int outs;
    private List<Runner> runners;
    
    // Getters and setters
    public String getGameId() { return gameId; }
    public void setGameId(String gameId) { this.gameId = gameId; }
    
    public int getInning() { return inning; }
    public void setInning(int inning) { this.inning = inning; }
    
    public String getInningState() { return inningState; }
    public void setInningState(String inningState) { this.inningState = inningState; }
    
    public int getHomeScore() { return homeScore; }
    public void setHomeScore(int homeScore) { this.homeScore = homeScore; }
    
    public int getAwayScore() { return awayScore; }
    public void setAwayScore(int awayScore) { this.awayScore = awayScore; }
    
    public int getOuts() { return outs; }
    public void setOuts(int outs) { this.outs = outs; }
    
    public List<Runner> getRunners() { return runners; }
    public void setRunners(List<Runner> runners) { this.runners = runners; }
    
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        GameState gameState = (GameState) o;
        return inning == gameState.inning &&
               homeScore == gameState.homeScore &&
               awayScore == gameState.awayScore &&
               outs == gameState.outs &&
               gameId.equals(gameState.gameId) &&
               inningState.equals(gameState.inningState);
    }
    
    @Override
    public int hashCode() {
        return gameId.hashCode() + inning + homeScore + awayScore + outs;
    }
