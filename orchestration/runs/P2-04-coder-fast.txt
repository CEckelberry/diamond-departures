"""
Abstract interfaces for Pub/Sub subscriber and cache invalidation.
Keeps implementation cloud-client agnostic.
"""

from abc import ABC, abstractmethod
from typing import Dict, Any, Tuple


class CacheInvalidationHook(ABC):
    """Interface for cache invalidation callbacks."""

    @abstractmethod
    def invalidate(self, view: str, sort: str) -> None:
        """
        Invalidate cache entries for a specific view and sort combination.

        Args:
            view: The view identifier.
            sort: The sort criteria identifier.
        """
        pass


class PubSubMessage(ABC):
    """Abstract representation of a Pub/Sub message."""

    @abstractmethod
    def get_payload(self) -> Dict[str, Any]:
        """Return the raw payload as a dictionary."""
        pass

    @abstractmethod
    def ack(self) -> None:
        """Acknowledge successful processing."""
        pass

    @abstractmethod
    def nack(self, requeue: bool = False) -> None:
        """
        Negative acknowledge, indicating failure.

        Args:
            requeue: Whether to requeue the message for retry.
        """
        pass
```

"""
Pub/Sub Subscriber implementation.
Consumes ingest refresh messages and triggers cache invalidation.
"""

import logging
from typing import Callable, Dict, Any, List, Optional

from .interfaces import PubSubMessage, CacheInvalidationHook

logger = logging.getLogger(__name__)


class PubSubSubscriber:
    """
    Subscribes to Pub/Sub topics, parses refresh messages,
    and triggers cache invalidation hooks.
    """

    def __init__(
        self,
        invalidation_hook: CacheInvalidationHook,
        message_handler: Optional[Callable[[PubSubMessage], None]] = None,
    ):
        """
        Initialize the subscriber.

        Args:
            invalidation_hook: The hook to call for cache invalidation.
            message_handler: Optional custom handler for raw messages.
        """
        self.invalidation_hook = invalidation_hook
        self.message_handler = message_handler

    def process_message(self, message: PubSubMessage) -> None:
        """
        Process a single Pub/Sub message.

        Args:
            message: The incoming Pub/Sub message.
        """
        try:
            payload = message.get_payload()
            view = payload.get("view")
            sort = payload.get("sort")

            if not view or not sort:
                raise ValueError("Missing 'view' or 'sort' in payload")

            # Log structured context
            logger.info(
                "Processing refresh message",
                extra={"view": view, "sort": sort, "reason": "ingest_refresh"},
            )

            # Trigger invalidation for each target key
            # In this scope, we assume one invalidation call per (view, sort)
            self.invalidation_hook.invalidate(view=view, sort=sort)

            # Acknowledge success
            message.ack()
            logger.info(
                "Message processed successfully",
                extra={"view": view, "sort": sort},
            )

        except Exception as e:
            logger.error(
                "Failed to process message",
                extra={
                    "reason": "processing_error",
                    "error": str(e),
                },
                exc_info=True,
            )
            # Negative acknowledge
            message.nack(requeue=False)

    def start(self) -> None:
        """Start the subscriber loop (placeholder for cloud-specific implementation)."""
        logger.info("Subscriber started")

    def stop(self) -> None:
        """Stop the subscriber loop."""
        logger.info("Subscriber stopped")
```

"""
Tests for Pub/Sub subscriber invalidation flow.
"""

import json
import logging
from typing import Dict, Any
from unittest.mock import Mock, patch

import pytest

from apps.api.pubsub.subscriber import PubSubSubscriber
from apps.api.pubsub.interfaces import PubSubMessage, CacheInvalidationHook
from apps.api.cache.invalidator import SimpleCacheInvalidator


class MockPubSubMessage(PubSubMessage):
    """Mock implementation of PubSubMessage for testing."""

    def __init__(self, payload: Dict[str, Any]):
        self._payload = payload
        self._acked = False
        self._nacked = False
        self._requeue = False

    def get_payload(self) -> Dict[str, Any]:
        return self._payload

    def ack(self) -> None:
        self._acked = True

    def nack(self, requeue: bool = False) -> None:
        self._nacked = True
        self._requeue = requeue

    @property
    def is_acked(self) -> bool:
        return self._acked

    @property
    def is_nacked(self) -> bool:
        return self._nacked

    @property
    def nack_requeue(self) -> bool:
        return self._requeue


class TestPubSubSubscriber:
    """Tests for PubSubSubscriber."""

    def test_process_message_success(self, caplog):
        """Test successful message processing with ACK."""
        hook = Mock(spec=CacheInvalidationHook)
        subscriber = PubSubSubscriber(invalidation_hook=hook)

        payload = {"view": "home", "sort": "date"}
        message = MockPubSubMessage(payload)

        with caplog.at_level(logging.INFO):
            subscriber.process_message(message)

        assert message.is_acked
        assert not message.is_nacked
        hook.invalidate.assert_called_once_with(view="home", sort="date")
        assert "Processing refresh message" in caplog.text
        assert "Message processed successfully" in caplog.text

    def test_process_message_missing_fields(self, caplog):
        """Test message processing with missing view or sort."""
        hook = Mock(spec=CacheInvalidationHook)
        subscriber = PubSubSubscriber(invalidation_hook=hook)

        payload = {"view": "home"}  # Missing 'sort'
        message = MockPubSubMessage(payload)

        with caplog.at_level(logging.ERROR):
            subscriber.process_message(message)

        assert not message.is_acked
        assert message.is_nacked
        assert not message.nack_requeue
        hook.invalidate.assert_not_called()
        assert "Failed to process message" in caplog.text
        assert "Missing 'view' or 'sort'" in caplog.text

    def test_process_message_invalid_json(self, caplog):
        """Test message processing with invalid payload structure."""
        hook = Mock(spec=CacheInvalidationHook)
        subscriber = PubSubSubscriber(invalidation_hook=hook)

        payload = {"view": None, "sort": "date"}  # view is None
        message = MockPubSubMessage(payload)

        with caplog.at_level(logging.ERROR):
            subscriber.process_message(message)

        assert not message.is_acked
        assert message.is_nacked
        hook.invalidate.assert_not_called()

    def test_structured_logging_context(self, caplog):
        """Test that structured logging includes view and sort."""
        hook = Mock(spec=CacheInvalidationHook)
        subscriber = PubSubSubscriber(invalidation_hook=hook)

        payload = {"view": "dashboard", "sort": "popularity"}
        message = MockPubSubMessage(payload)

        with caplog.at_level(logging.INFO):
            subscriber.process_message(message)

        # Check if the log record contains the expected extra fields
        assert any(
            record.view == "dashboard" and record.sort == "popularity"
            for record in caplog.records
        )
