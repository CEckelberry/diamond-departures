```python
"""Player reader service for fetching player details and history."""

from typing import List, Optional
from uuid import UUID

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from apps.api.models.player import Player
from apps.api.models.history import PlayerHistory


class PlayerReader:
    """Reads player data from the database."""

    def __init__(self, session: AsyncSession):
        self.session = session

    async def get_player(self, player_id: UUID) -> Optional[Player]:
        """Fetch a player by ID."""
        result = await self.session.execute(select(Player).where(Player.id == player_id))
        return result.scalar_one_or_none()

    async def get_player_history(
        self, player_id: UUID, stat_type: Optional[str] = None
    ) -> List[PlayerHistory]:
        """Fetch player history, optionally filtered by stat type."""
        query = select(PlayerHistory).where(PlayerHistory.player_id == player_id)
        if stat_type:
            query = query.where(PlayerHistory.stat_type == stat_type)
        query = query.order_by(PlayerHistory.timestamp.asc())
        result = await self.session.execute(query)
        return list(result.scalars().all())
```

```python
"""Player detail and history endpoints."""

from typing import Optional
from uuid import UUID

from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.ext.asyncio import AsyncSession

from apps.api.db.session import get_db_session
from apps.api.schemas.player import PlayerDetail, PlayerHistoryEntry, PlayerHistoryResponse
from apps.api.services.player_reader import PlayerReader

router = APIRouter(prefix="/api/players", tags=["players"])


@router.get("/{player_id}", response_model=PlayerDetail)
async def get_player_detail(
    player_id: UUID,
    db_session: AsyncSession = Depends(get_db_session),
):
    """Get player detail by ID."""
    reader = PlayerReader(db_session)
    player = await reader.get_player(player_id)
    if not player:
        raise HTTPException(status_code=404, detail="Player not found")
    return player


@router.get("/{player_id}/history", response_model=PlayerHistoryResponse)
async def get_player_history(
    player_id: UUID,
    stat_type: Optional[str] = Query(None, description="Filter by stat type"),
    db_session: AsyncSession = Depends(get_db_session),
):
    """Get player history, optionally filtered by stat type."""
    reader = PlayerReader(db_session)
    player = await reader.get_player(player_id)
    if not player:
        raise HTTPException(status_code=404, detail="Player not found")

    history = await reader.get_player_history(player_id, stat_type)

    # If stat_type is provided but no history found, it's not a 400 error per se,

    # Re-reading scope: "400 for invalid/missing history stat".
```
