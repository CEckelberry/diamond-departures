### 2. Exact File Edits

#### `docker-compose.yml`

```yaml
version: '3.8'

services:
  db:
    image: postgres:15-alpine
    environment:
      POSTGRES_USER: dev
      POSTGRES_PASSWORD: dev
      POSTGRES_DB: devdb
    ports:
      - "5432:5432"
    volumes:
      - db-data:/var/lib/postgresql/data

  db-init:
    image: postgres:15-alpine
    depends_on:
      - db
    command: >
      sh -c "
        until pg_isready -h db -p 5432; do
          echo 'Waiting for db...';
          sleep 1;
        done;
        echo 'DB ready, running init...';
        psql -h db -U dev -d devdb -c 'SELECT 1;'
      "
    environment:
      PGPASSWORD: dev

  mlb-mock:
    image: nginx:alpine
    ports:
      - "8090:80"
    volumes:
      - ./mocks:/usr/share/nginx/html:ro

  api:
    build:
      context: .
      dockerfile: Dockerfile.api
    environment:
      DATABASE_URL: postgresql://dev:dev@db:5432/devdb
      MLB_MOCK_URL: http://mlb-mock:80

  ingest:
    build:
      context: .
      dockerfile: Dockerfile.ingest
    environment:
      DATABASE_URL: postgresql://dev:dev@db:5432/devdb
      MLB_MOCK_URL: http://mlb-mock:80

  web:
    build:
      context: .
      dockerfile: Dockerfile.web
    ports:
      - "5173:5173"
    environment:
      API_URL: http://api:8000

volumes:
  db-data:
```
