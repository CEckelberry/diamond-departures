# Packet P7-04 — SEO + Open Graph GREEN

## 1. Plan

### Assumptions
1.  **Framework**: The project uses a standard Node.js/TypeScript stack with `express` (or compatible middleware) and `react` (or compatible SSR/SSG setup). I will assume a standard `src/routes` and `src/components` structure.
2.  **Dependencies**: `sharp` is available for image processing, or we will use a lightweight placeholder strategy if `sharp` is too heavy for this specific "baseline" packet. Given the "placeholder" requirement, I will implement a simple SVG-based OG image generator to avoid heavy binary dependencies for now, ensuring the route works immediately.
3.  **Routing**: We will add specific routes for `/robots.txt`, `/sitemap.xml`, and `/og-image.png`.
4.  **Metadata**: We will create a context or utility to manage page-level metadata (title, description, OG tags) that can be consumed by the layout component.

### Risks & Failure Modes
1.  **Caching**: Bots (Google, Twitter) cache OG images aggressively. We must ensure proper `Cache-Control` headers are set on the OG image route.
2.  **Image Dimensions**: OG images must be exactly 1200x630 pixels for optimal display. The generator must enforce this.
3.  **SSR/SSG Integration**: If using Next.js or similar, metadata is often handled via `next/head` or config files. Since this is a "baseline" implementation, I will assume a custom React/Express setup where we inject metadata into the HTML head. If this were Next.js, the approach would differ significantly (using `next-seo` or built-in metadata API). *Decision*: I will implement a generic `MetadataProvider` and route handlers that work in a standard SSR/Express environment.
4.  **Sitemap Validity**: The sitemap must be valid XML. Special characters in URLs must be escaped.

### Implementation Steps
1.  **Metadata Context**: Create a `MetadataContext` to allow components to define their SEO metadata.
2.  **OG Image Route**: Create `/api/og` or `/og-image.png` that generates a PNG from an SVG template.
3.  **Robots.txt Route**: Serve a static `robots.txt` file.
4.  **Sitemap Route**: Create a dynamic sitemap generator that lists all known pages.
5.  **Integration**: Wire these into the main app router and ensure the layout component consumes the metadata context to inject `<meta>` tags.

### File: `src/context/MetadataContext.tsx`
```typescript
import React, { createContext, useContext, useState, ReactNode } from 'react';

export interface PageMetadata {
  title?: string;
  description?: string;
  ogImage?: string;
  canonicalUrl?: string;
}

interface MetadataContextType {
  metadata: PageMetadata;
  setMetadata: (meta: Partial<PageMetadata>) => void;
}

const MetadataContext = createContext<MetadataContextType | undefined>(undefined);

export const MetadataProvider: React.FC<{ children: ReactNode; initialMetadata?: PageMetadata }> = ({ 
  children, 
  initialMetadata = {} 
}) => {
  const [metadata, setMetadataState] = useState<PageMetadata>(initialMetadata);

  const setMetadata = (partial: Partial<PageMetadata>) => {
    setMetadataState((prev) => ({ ...prev, ...partial }));
  };

  return (
    <MetadataContext.Provider value={{ metadata, setMetadata }}>
      {children}
    </MetadataContext.Provider>
  );
};

export const useMetadata = () => {
  const context = useContext(MetadataContext);
  if (!context) {
    throw new Error('useMetadata must be used within a MetadataProvider');
  }
  return context;
};
```

### File: `src/components/Head.tsx`
```typescript
import React, { useEffect } from 'react';
import { useMetadata } from '../context/MetadataContext';

const DEFAULT_TITLE = 'My App';
const DEFAULT_DESCRIPTION = 'Welcome to My App';
const DEFAULT_OG_IMAGE = '/og-image.png'; // Placeholder route

export const Head: React.FC = () => {
  const { metadata } = useMetadata();

  useEffect(() => {
    // Update document title
    document.title = metadata.title || DEFAULT_TITLE;

    // Update meta tags
    const setMeta = (name: string, content: string, attr: string = 'name') => {
      let el = document.querySelector(`meta[${attr}="${name}"]`);
      if (!el) {
        el = document.createElement('meta');
        el.setAttribute(attr, name);
        document.head.appendChild(el);
      }
      el.setAttribute('content', content);
    };

    setMeta('description', metadata.description || DEFAULT_DESCRIPTION);
    setMeta('og:title', metadata.title || DEFAULT_TITLE, 'property');
    setMeta('og:description', metadata.description || DEFAULT_DESCRIPTION, 'property');
    setMeta('og:image', metadata.ogImage || DEFAULT_OG_IMAGE, 'property');
    setMeta('og:type', 'website', 'property');
    setMeta('twitter:card', 'summary_large_image', 'name');
    
    // Canonical URL
    if (metadata.canonicalUrl) {
      let link = document.querySelector('link[rel="canonical"]');
      if (!link) {
        link = document.createElement('link');
        link.setAttribute('rel', 'canonical');
        document.head.appendChild(link);
      }
      link.setAttribute('href', metadata.canonicalUrl);
    }
  }, [metadata]);

  return null;
};
```

### File: `src/routes/seo.ts`
```typescript
import express, { Request, Response } from 'express';
import { createCanvas } from 'canvas'; // Assuming 'canvas' package is used for server-side image gen
// Note: If 'sharp' is preferred, replace this block with sharp logic.
// For this baseline, we'll use a simple SVG-to-PNG approach if 'canvas' is available,
// or return a static placeholder if not. Let's assume 'canvas' is installed for simplicity.

const router = express.Router();

// 1. Robots.txt
router.get('/robots.txt', (req: Request, res: Response) => {
  const robotsTxt = `
User-agent: *
Allow: /
Sitemap: ${process.env.BASE_URL || 'http://localhost:3000'}/sitemap.xml
  `.trim();
  
  res.set('Content-Type', 'text/plain');
  res.send(robotsTxt);
});

// 2. Sitemap.xml
router.get('/sitemap.xml', (req: Request, res: Response) => {
  // In a real app, this would query the database for all pages.
  // For baseline, we hardcode a few known routes.
  const baseUrl = process.env.BASE_URL || 'http://localhost:3000';
  const pages = [
    { loc: '/', lastmod: new Date().toISOString(), changefreq: 'daily', priority: '1.0' },
    { loc: '/about', lastmod: new Date().toISOString(), changefreq: 'monthly', priority: '0.8' },
    { loc: '/contact', lastmod: new Date().toISOString(), changefreq: 'monthly', priority: '0.5' },
  ];

  const xml = `<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
${pages.map(page => `
  <url>
    <loc>${baseUrl}${page.loc}</loc>
    <lastmod>${page.lastmod}</lastmod>
    <changefreq>${page.changefreq}</changefreq>
    <priority>${page.priority}</priority>
  </url>`).join('')}
</urlset>`;

  res.set('Content-Type', 'application/xml');
  res.send(xml);
});

// 3. OG Image Placeholder
// Generates a simple PNG with text overlay.
router.get('/og-image.png', (req: Request, res: Response) => {
  const width = 1200;
  const height = 630;
  
  // Create a simple SVG as a placeholder for the OG image
  // In production, this would be a complex template with branding.
  const svg = `
    <svg xmlns="http://www.w3.org/2000/svg" width="${width}" height="${height}">
      <rect width="100%" height="100%" fill="#1a1a1a"/>
      <text x="50%" y="50%" dominant-baseline="middle" text-anchor="middle" font-family="sans-serif" font-size="60" fill="#ffffff">
        My App
      </text>
      <text x="50%" y="60%" dominant-baseline="middle" text-anchor="middle" font-family="sans-serif" font-size="30" fill="#cccccc">
        Open Graph Image Placeholder
      </text>
    </svg>
  `;

  // Convert SVG to PNG using canvas
  const canvas = createCanvas(width, height);
  const ctx = canvas.getContext('2d');
  
  // Note: 'canvas' package doesn't natively support SVG rendering without additional libraries like 'svg-canvas'.
  // For this baseline, we will draw a simple rectangle and text to avoid dependency hell.
  // This is a "placeholder" as per requirements.
  
  ctx.fillStyle = '#1a1a1a';
  ctx.fillRect(0, 0, width, height);
  
  ctx.fillStyle = '#ffffff';
  ctx.font = '60px sans-serif';
  ctx.textAlign = 'center';
  ctx.fillText('My App', width / 2, height / 2 - 30);
  
  ctx.font = '30px sans-serif';
  ctx.fillStyle = '#cccccc';
  ctx.fillText('Open Graph Image Placeholder', width / 2, height / 2 + 30);

  const buffer = canvas.toBuffer('image/png');
  
  res.set('Content-Type', 'image/png');
  res.set('Cache-Control', 'public, max-age=86400'); // Cache for 24 hours
  res.send(buffer);
});

export default router;
```
