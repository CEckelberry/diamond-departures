### File: `packages/stats/verify/harness.py`
```python
"""
Verification harness for stats package.
Loads data.json, computes stats, and compares against expected ground truth.
"""

import json
import math
import os
from typing import Any, Dict

# Import stats functions from the main stats package
# Assuming the stats package is installed or in path
try:
    from packages.stats import compute_stats
except ImportError:
    # Fallback for direct execution if not installed
    import sys
    sys.path.insert(0, os.path.join(os.path.dirname(__file__), '..', '..'))
    from packages.stats import compute_stats

TOLERANCE = 0.005  # 0.5%


def load_data(filepath: str) -> Dict[str, Any]:
    """Load dataset and expected stats from JSON file."""
    with open(filepath, 'r') as f:
        return json.load(f)


def compute_expected_stats(data: list) -> Dict[str, float]:
    """
    Compute stats using the harness's own logic to ensure independence.
    In a real scenario, this might just call the library, but here we
    verify the library against known mathematical truths.
    """
    n = len(data)
    if n == 0:
        raise ValueError("Empty dataset")
    
    mean = sum(data) / n
    variance = sum((x - mean) ** 2 for x in data) / n
    std_dev = math.sqrt(variance)
    sorted_data = sorted(data)
    
    if n % 2 == 0:
        median = (sorted_data[n // 2 - 1] + sorted_data[n // 2]) / 2
    else:
        median = sorted_data[n // 2]
        
    return {
        "mean": mean,
        "variance": variance,
        "std_dev": std_dev,
        "min": min(data),
        "max": max(data),
        "count": n,
        "median": median
    }


def verify_stats(filepath: str) -> bool:
    """
    Verify computed stats against expected ground truth.
    Returns True if all checks pass within tolerance.
    """
    data = load_data(filepath)
    dataset = data["dataset"]
    expected = data["expected_stats"]
    
    # Compute stats using the library
    computed = compute_stats(dataset)
    
    errors = []
    for key, exp_val in expected.items():
        if key not in computed:
            errors.append(f"Missing key '{key}' in computed stats")
            continue
            
        comp_val = computed[key]
        
        # Handle float comparison with tolerance
        if isinstance(exp_val, float):
            if abs(comp_val - exp_val) > abs(exp_val) * TOLERANCE:
                errors.append(
                    f"Mismatch for '{key}': expected {exp_val}, got {comp_val} "
                    f"(diff: {abs(comp_val - exp_val)}, tolerance: {abs(exp_val) * TOLERANCE})"
                )
        else:
            if comp_val != exp_val:
                errors.append(
                    f"Mismatch for '{key}': expected {exp_val}, got {comp_val}"
                )
                
    if errors:
        print("Verification FAILED:")
        for err in errors:
            print(f"  - {err}")
        return False
    else:
        print("Verification PASSED: All stats within 0.5% tolerance.")
        return True


if __name__ == "__main__":
    script_dir = os.path.dirname(os.path.abspath(__file__))
    data_path = os.path.join(script_dir, "data.json")
    success = verify_stats(data_path)
    exit(0 if success else 1)
```

"""
Tests for the verification harness.
"""

import json
import os
import pytest
from unittest.mock import patch, MagicMock

# Import harness functions
import sys
sys.path.insert(0, os.path.join(os.path.dirname(__file__), '..'))
from verify.harness import load_data, verify_stats, TOLERANCE


class TestLoadData:
    def test_load_data_success(self, tmp_path):
        """Test that data.json can be loaded correctly."""
        data_file = tmp_path / "data.json"
        data_file.write_text(json.dumps({
            "dataset": [1, 2, 3],
            "expected_stats": {"mean": 2.0}
        }))
        
        result = load_data(str(data_file))
        assert result["dataset"] == [1, 2, 3]
        assert result["expected_stats"]["mean"] == 2.0

    def test_load_data_missing_file(self):
        """Test that missing file raises FileNotFoundError."""
        with pytest.raises(FileNotFoundError):
            load_data("/nonexistent/path/data.json")


class TestVerifyStats:
    @patch('verify.harness.compute_stats')
    def test_verify_stats_passes(self, mock_compute_stats, tmp_path):
        """Test that verification passes when computed stats match expected."""
        # Setup mock to return expected values
        mock_compute_stats.return_value = {
            "mean": 50.5,
            "variance": 833.25,
            "std_dev": 28.866084,
            "min": 1,
            "max": 100,
            "count": 100,
            "median": 50.5
        }
        
        # Create a test data file
        data_file = tmp_path / "test_data.json"
        data_file.write_text(json.dumps({
            "dataset": list(range(1, 101)),
            "expected_stats": {
                "mean": 50.5,
                "variance": 833.25,
                "std_dev": 28.866084,
                "min": 1,
                "max": 100,
                "count": 100,
                "median": 50.5
            }
        }))
        
        result = verify_stats(str(data_file))
        assert result is True

    @patch('verify.harness.compute_stats')
    def test_verify_stats_fails_on_mismatch(self, mock_compute_stats, tmp_path):
        """Test that verification fails when computed stats differ significantly."""
        # Setup mock to return incorrect values
        mock_compute_stats.return_value = {
            "mean": 100.0,  # Way off from 50.5
            "variance": 833.25,
            "std_dev": 28.866084,
            "min": 1,

            "count": 100,
            "median": 50.5
        }
        
        data_file = tmp_path / "test_data.json"
        data_file.write_text(json.dumps({
            "dataset": list(range(1, 101)),
            "expected_stats": {
                "mean": 50.5,
                "variance": 833.25,
                "std_dev": 28.866084,
                "min": 1,
                "max": 100,
                "count": 100,
                "median": 50.5
            }
        }))
        
        result = verify_stats(str(data_file))
        assert result is False

    @patch('verify.harness.compute_stats')
    def test_verify_stats_fails_on_missing_key(self, mock_compute_stats, tmp_path):
        """Test that verification fails when a key is missing in computed stats."""
