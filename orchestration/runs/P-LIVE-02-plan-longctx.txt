I'll implement the scanner checkpoint/cursor reliability feature. Let me break this down into a clear implementation plan.

## Implementation Plan

### Assumptions
- The scanner currently processes game events in a loop without persistence
- We need to track which games have been processed to avoid duplicates
- JSON will be used for checkpoint storage (simple, portable, human-readable)
- Checkpoints should be written after successful game processing
- Metrics should be collected per scanner run

### Risks
- **Risk**: Checkpoint corruption could cause duplicate processing
  - *Mitigation*: Implement atomic write with temp file + rename
- **Risk**: Checkpoint file corruption on crash
  - *Mitigation*: Add checksum validation on read
- **Risk**: Metrics collection impacting performance
  - *Mitigation*: Use lightweight counters, batch metrics output

### Edge Cases & Failure Modes
1. **Checkpoint file doesn't exist**: Scanner should start from beginning
2. **Invalid checkpoint format**: Scanner should fall back to beginning
3. **Permission issues**: Scanner should log error and continue
4. **Concurrent access**: Need to handle file locking
5. **Disk full**: Should catch write errors and log appropriately

# scanner/checkpoint.py
import json
import os
import logging
from datetime import datetime
from pathlib import Path
from typing import Optional, Dict, Any

logger = logging.getLogger(__name__)

class CheckpointStore:
    """Durable checkpoint storage for game scanner."""
    
    def __init__(self, checkpoint_dir: str = "checkpoints"):
        self.checkpoint_dir = Path(checkpoint_dir)
        self.checkpoint_dir.mkdir(exist_ok=True)
        self.checkpoint_file = self.checkpoint_dir / "scanner_checkpoint.json"
        
    def save_checkpoint(self, game_id: str, processed_at: Optional[datetime] = None) -> None:
        """Save checkpoint for a game."""
        try:
            checkpoint_data = self._load_checkpoint()
            checkpoint_data[game_id] = {
                "processed_at": processed_at or datetime.utcnow().isoformat(),
                "game_id": game_id
            }
            self._write_checkpoint(checkpoint_data)
            logger.debug(f"Saved checkpoint for game {game_id}")
        except Exception as e:
            logger.error(f"Failed to save checkpoint for game {game_id}: {e}")
            raise
    
    def get_checkpoint(self, game_id: str) -> Optional[Dict[str, Any]]:
        """Get checkpoint for a specific game."""
        try:
            checkpoint_data = self._load_checkpoint()
            return checkpoint_data.get(game_id)
        except Exception as e:
            logger.error(f"Failed to read checkpoint for game {game_id}: {e}")
            return None
    
    def get_all_checkpoints(self) -> Dict[str, Dict[str, Any]]:
        """Get all checkpoints."""
        try:
            return self._load_checkpoint()
        except Exception as e:
            logger.error(f"Failed to read all checkpoints: {e}")
            return {}
    
    def clear_checkpoint(self, game_id: str) -> None:
        """Clear checkpoint for a game."""
        try:
            checkpoint_data = self._load_checkpoint()
            if game_id in checkpoint_data:
                del checkpoint_data[game_id]
                self._write_checkpoint(checkpoint_data)
                logger.debug(f"Cleared checkpoint for game {game_id}")
        except Exception as e:
            logger.error(f"Failed to clear checkpoint for game {game_id}: {e}")
    
    def _load_checkpoint(self) -> Dict[str, Any]:
        """Load checkpoint data from file."""
        if not self.checkpoint_file.exists():
            return {}
        
        try:
            with open(self.checkpoint_file, 'r') as f:
                return json.load(f)
        except (json.JSONDecodeError, IOError) as e:
            logger.warning(f"Invalid checkpoint file format: {e}")
            return {}
    
    def _write_checkpoint(self, data: Dict[str, Any]) -> None:
        """Write checkpoint data to file atomically."""
        temp_file = self.checkpoint_file.with_suffix('.tmp')
        
        try:
            with open(temp_file, 'w') as f:
                json.dump(data, f, indent=2)
            # Atomic rename
            temp_file.replace(self.checkpoint_file)
        except IOError as e:
            logger.error(f"Failed to write checkpoint file: {e}")
            raise

# scanner/scanner.py
import logging
from typing import List, Dict, Any, Optional
from .metrics import ScannerMetrics
from .game_loader import GameLoader

logger = logging.getLogger(__name__)

class GameScanner:
    def __init__(self, checkpoint_store: Optional[CheckpointStore] = None):
        self.checkpoint_store = checkpoint_store or CheckpointStore()
        self.metrics = ScannerMetrics()
        self.game_loader = GameLoader()
    
    def scan(self, game_ids: List[str]) -> Dict[str, Any]:
        """Scan games and return results with metrics."""
        results = {
            'processed_games': 0,
            'failed_games': 0,
            'changed_games': 0,
            'feed_failures': 0,
            'processing_time': 0,
            'details': []
        }
        
        start_time = self.metrics.now()
        
        for game_id in game_ids:
            try:
                # Check if game was already processed
                last_checkpoint = self.checkpoint_store.get_checkpoint(game_id)
                
                # Process game
                game_result = self._process_game(game_id, last_checkpoint)
                
                # Update metrics
                if game_result['changed']:
                    results['changed_games'] += 1
                if game_result['feed_failure']:
                    results['feed_failures'] += 1
                
                # Save checkpoint on success
                if not game_result['error']:
                    self.checkpoint_store.save_checkpoint(game_id)
                
                results['details'].append(game_result)
                results['processed_games'] += 1
                
            except Exception as e:
                logger.error(f"Error processing game {game_id}: {e}")
                results['failed_games'] += 1
                results['details'].append({
                    'game_id': game_id,
                    'error': str(e),
                    'error_type': type(e).__name__
                })
        
        results['processing_time'] = self.metrics.now() - start_time
        return results
    
    def _process_game(self, game_id: str, last_checkpoint: Optional[Dict[str, Any]]) -> Dict[str, Any]:
        """Process a single game and return result."""
        start_time = self.metrics.now()
        
        try:
            # Load game data
            game_data = self.game_loader.load_game(game_id)
            
            # Check if data changed since last checkpoint
            changed = self._has_data_changed(game_data, last_checkpoint)
            
            # Process game feed
            feed_result = self.game_loader.process_feed(game_data)
            
            return {
                'game_id': game_id,
                'changed': changed,
                'feed_failure': feed_result.get('failure', False),
                'processing_time': self.metrics.now() - start_time,
                'error': False
            }
            
        except Exception as e:
            return {
                'game_id': game_id,
                'error': True,
                'error_message': str(e),
                'processing_time': self.metrics.now() - start_time
            }
    
    def _has_data_changed(self, current_data: Dict[str, Any], last_checkpoint: Optional[Dict[str, Any]]) -> bool:
        """Check if game data has changed since last checkpoint."""
        if not last_checkpoint:
            return True
        
        # Compare relevant data fields
        current_hash = hash(str(current_data))
        checkpoint_hash = hash(str(last_checkpoint.get('data', {})))
        
        return current_hash != checkpoint_hash

# scanner/metrics.py
from typing import Dict, Any, List
from datetime import datetime
import time

class ScannerMetrics:
    def __init__(self):
        self.start_time = None
        self.end_time = None
        self.game_processing_times = []
        self.feed_failure_count = 0
        self.checkpoint_write_count = 0
        self.checkpoint_read_count = 0
        self.errors = []
    
    def now(self) -> float:
        """Return current time in seconds."""
        return time.time()
    
    def start(self) -> None:
        """Start timing."""
        self.start_time = self.now()
    
    def end(self) -> None:
        """End timing."""
        self.end_time = self.now()
    
    def record_game_processing(self, game_id: str, processing_time: float, changed: bool, feed_failure: bool) -> None:
        """Record metrics for a single game."""
        self.game_processing_times.append({
            'game_id': game_id,
            'processing_time': processing_time,
            'changed': changed,
            'feed_failure': feed_failure,
            'timestamp': datetime.utcnow().isoformat()
        })
    
    def record_checkpoint_write(self) -> None:
        """Record checkpoint write operation."""
        self.checkpoint_write_count += 1
    
    def record_checkpoint_read(self) -> None:
        """Record checkpoint read operation."""
        self.checkpoint_read_count += 1
    
    def record_error(self, error: str, error_type: str = "unknown") -> None:
        """Record an error."""
        self.errors.append({
            'error': error,
            'error_type': error_type,
            'timestamp': datetime.utcnow().isoformat()
        })
    
    def get_summary(self) -> Dict[str, Any]:
        """Get summary metrics."""
        if not self.start_time:
            return {}
        
        total_processing_time = self.end_time - self.start_time if self.end_time else 0
        avg_processing_time = sum(self.game_processing_times, {}).get('processing_time', 0) / len(self.game_processing_times) if self.game_processing_times else 0
        
        return {
            'total_processing_time': total_processing_time,
            'games_processed': len(self.game_processing_times),
            'avg_processing_time': avg_processing_time,
            'feed_failures': self.feed_failure_count,
            'checkpoint_writes': self.checkpoint_write_count,
            'checkpoint_reads': self.checkpoint_read_count,
            'errors': len(self.errors),
            'error_details': self.errors
        }
