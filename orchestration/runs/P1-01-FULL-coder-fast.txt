+"""Weighted On-Base Average (wOBA) implementation."""
+
+import json
+from dataclasses import dataclass
+from pathlib import Path
+
+
+@dataclass
+class WOBAWeights:
+    """Weights for wOBA calculation components."""
+
+    bb: float
+    hbp: float
+    single: float
+    double: float
+    triple: float
+    hr: float
+
+
+def load_woba_weights(year: int = 2026) -> WOBAWeights:
+    """Load wOBA weights for a specific year from JSON.
+
+    Args:
+        year: The year to load weights for.
+
+    Returns:
+        WOBAWeights instance.
+
+    Raises:
+        FileNotFoundError: If weights file does not exist.
+        ValueError: If weights file is invalid.
+    """
+    weights_path = Path(__file__).parent / f"woba_weights_{year}.json"
+    with open(weights_path, "r") as f:
+        data = json.load(f)
+
+    return WOBAWeights(
+        bb=data["bb"],
+        hbp=data["hbp"],
+        single=data["single"],
+        double=data["double"],
+        triple=data["triple"],
+        hr=data["hr"],
+    )
+
+
+def woba(
+    bb: float,
+    hbp: float,
+    single: float,
+    double: float,
+    triple: float,
+    hr: float,
+    ab: float,
+    sf: float,
+    ibb: float = 0.0,
+) -> float:
+    """Calculate Weighted On-Base Average (wOBA).
+
+    Formula:
+        (0.69*BB + 0.72*HBP + 0.89*1B + 1.27*2B + 1.62*3B + 2.10*HR) /
+        (AB + BB - IBB + SF + HBP)
+
+    Args:
+        bb: Walks.
+        hbp: Hit by pitch.
+        single: Singles.
+        double: Doubles.
+        triple: Triples.
+        hr: Home runs.
+        ab: At-bats.
+        sf: Sacrifice flies.
+        ibb: Intentional walks (default 0).
+
+    Returns:
+        wOBA value.
+
+    Raises:
+        ZeroDivisionError: If denominator is zero.
+    """
+    numerator = (
+        0.69 * bb
+        + 0.72 * hbp
+        + 0.89 * single
+        + 1.27 * double
+        + 1.62 * triple
+        + 2.10 * hr
+    )
+    denominator = ab + bb - ibb + sf + hbp
+
+    if denominator == 0:
+        raise ZeroDivisionError("Denominator for wOBA is zero")
+
+    return numerator / denominator

+"""Weighted Runs Created Plus (wRC+) implementation."""
+
+from dataclasses import dataclass
+
+from .woba import woba, WOBAWeights
+
+
+@dataclass
+class LeagueContext:
+    """Contextual league statistics for adjustment.
+
+    Attributes:
+        league_woba: League average wOBA.
+        woba_scale: Scale factor for wOBA.
+        league_runs_per_pa: League runs per plate appearance.
+        park_factor: Park factor (1.0 is neutral).
+    """
+
+    league_woba: float
+    woba_scale: float
+    league_runs_per_pa: float
+    park_factor: float
+
+
+def wrc_plus(
+    bb: float,
+    hbp: float,
+    single: float,
+    double: float,
+    triple: float,
+    hr: float,
+    ab: float,
+    sf: float,
+    ibb: float = 0.0,
+    pa: float = 0.0,
+    context: LeagueContext | None = None,
+    weights: WOBAWeights | None = None,
+) -> float:
+    """Calculate Weighted Runs Created Plus (wRC+).
+
+    Formula:
+        100 * ((wOBA - lgwOBA) / wOBAScale + lgRuns/PA + (lgRuns/PA - PF * lgRuns/PA))
+        / (lgRuns/PA)
+
+    Note: If pa is 0, it is calculated as ab + bb - ibb + sf + hbp.
+
+    Args:
+        bb: Walks.
+        hbp: Hit by pitch.
+        single: Singles.
+        double: Doubles.
+        triple: Triples.
+        hr: Home runs.
+        ab: At-bats.
+        sf: Sacrifice flies.
+        ibb: Intentional walks (default 0).
+        pa: Plate appearances (optional, calculated if 0).
+        context: League context (optional).
+        weights: wOBA weights (optional).
+
+    Returns:
+        wRC+ value.
+
+    Raises:
+        ZeroDivisionError: If league runs per PA is zero.
+    """
+    if context is None:
+        # Default context for testing/defaults
+        context = LeagueContext(
+            league_woba=0.320,
+            woba_scale=0.72,
+            league_runs_per_pa=0.45,
+            park_factor=1.0,
+        )
+
+    if weights is None:
+        from .woba import load_woba_weights
+
+        weights = load_woba_weights()
+
+    # Calculate wOBA
+    player_woba = woba(
+        bb=bb,
+        hbp=hbp,
+        single=single,
+        double=double,
+        triple=triple,
+        hr=hr,
+        ab=ab,
+        sf=sf,
+        ibb=ibb,
+    )
+
+    # Calculate PA if not provided
+    if pa == 0:
+        pa = ab + bb - ibb + sf + hbp
+
+    lg_woba = context.league_woba
+    woba_scale = context.woba_scale
+    lg_rpa = context.league_runs_per_pa
+    pf = context.park_factor
+
+    if lg_rpa == 0:
+        raise ZeroDivisionError("League runs per PA cannot be zero")
+
+    # Formula: 100 * ((wOBA - lgwOBA) / wOBAScale + lgRuns/PA + (lgRuns/PA - PF * lgRuns/PA)) / (lgRuns/PA)
+    # Simplified: 100 * ((wOBA - lgwOBA) / wOBAScale + lgRuns/PA * (1 + 1 - PF)) / (lgRuns/PA)
+    # Actually, let's stick to the literal formula provided:
+    # 100 * ( (wOBA - lgwOBA) / wOBAScale + lgRuns/PA + (lgRuns/PA - PF * lgRuns/PA) ) / (lgRuns/PA)
+    
+    numerator = (
+        (player_woba - lg_woba) / woba_scale
+        + lg_rpa
+        + (lg_rpa - pf * lg_rpa)
+    )
+    
+    return 100 * (numerator / lg_rpa)

+        # Num: 0.69*1 + 0.72*1 + 0.89*1 + 2.10*1 = 4.4
