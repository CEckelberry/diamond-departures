### `apps/mlb-mock/app/main.py`
```python
import json
import os
from datetime import datetime, timedelta
from typing import Optional

from fastapi import FastAPI, Query
from fastapi.responses import JSONResponse

app = FastAPI(title="MLB Mock API")

DATA_DIR = os.path.join(os.path.dirname(__file__), "data")

def load_json(filename: str):
    filepath = os.path.join(DATA_DIR, filename)
    with open(filepath, "r") as f:
        return json.load(f)

@app.get("/api/v1/schedule")
async def get_schedule(date: str = Query("today", description="Date in YYYY-MM-DD or 'today'")):
    schedule = load_json("schedule.json")
    
    if date == "today":
        # For mock purposes, assume today is 2023-04-01
        target_date = "2023-04-01"
    else:
        target_date = date
    
    filtered = [game for game in schedule if game["date"] == target_date]
    return {"games": filtered, "date": target_date}

@app.get("/api/v1/game-feed")
async def get_game_feed(
    game_id: str = Query(..., description="Game ID"),
    speed: float = Query(1.0, description="Replay speed multiplier")
):
    feed_data = load_json("game_feed.json")
    game_feed = next((g for g in feed_data if g["game_id"] == game_id), None)
    
    if not game_feed:
        return JSONResponse(status_code=404, content={"error": "Game not found"})
    
    events = game_feed["events"]
    
    # Replay speed behavior: simulate time progression
    # If speed > 1, we pretend more time has passed. 
    # For this mock, we'll just return all events if speed >= 1, 
    # or filter by a simulated current time if speed < 1 (not implemented for simplicity).
    # To demonstrate "time-progressing feed markers", we add a 'simulated_current_time' field.
    
    base_time = datetime.fromisoformat(events[0]["time"].replace("Z", "+00:00")) if events else datetime.now()
    simulated_current_time = base_time + timedelta(hours=9 * speed)  # Assume 9 hour game
    
    return {
        "game_id": game_id,
        "simulated_current_time": simulated_current_time.isoformat(),
        "events": events
    }

@app.get("/api/v1/people/stats")
async def get_people_stats(player_id: str = Query(..., description="Player ID")):
    stats_data = load_json("people_stats.json")
    player = next((p for p in stats_data if p["player_id"] == player_id), None)
    
    if not player:
        return JSONResponse(status_code=404, content={"error": "Player not found"})
    
    return {
        "player_id": player["player_id"],
        "name": player["name"],
        "team": player["team"],
        "stats": player["stats"]
    }

@app.get("/api/v1/roster")
async def get_roster(team_id: str = Query(..., description="Team ID")):
    roster_data = load_json("roster.json")
    team_roster = next((t for t in roster_data if t["team_id"] == team_id), None)
    
    if not team_roster:
        return JSONResponse(status_code=404, content={"error": "Team not found"})
    
    return {
        "team_id": team_id,
        "roster": team_roster["roster"]
    }
```
