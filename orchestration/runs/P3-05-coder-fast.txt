<script lang="ts">
	/**
	 * Flap Component Test Page
	 * Interactive tuning page for cell/word/sound scenarios.
	 * Gated behind dev flag.
	 */

	// Dev gate: Only render in development
	if (!import.meta.env.DEV) {
		throw new Error('Flap test page is only available in development mode.');
	}

	// State
	let rows = 10;
	let cols = 10;
	let grid: boolean[][] = Array.from({ length: rows }, () => Array(cols).fill(false));
	let message = 'Ready to flap.';

	// Word list for word flip scenario
	const words = ['FLAP', 'GRID', 'TEST', 'DEV', 'CODE'];
	let currentWordIndex = 0;

	/**
	 * Flip a single cell at (r, c)
	 */
	function flipCell(r: number, c: number) {
		if (r >= 0 && r < rows && c >= 0 && c < cols) {
			grid[r][c] = !grid[r][c];
			message = `Flipped cell (${r}, ${c})`;
		}
	}

	/**
	 * Flip all cells in a "word" (simulated as a row segment)
	 */
	function flipWord() {
		const word = words[currentWordIndex % words.length];
		const startCol = currentWordIndex * 2; // Spread words out
		const length = word.length;

		for (let i = 0; i < length; i++) {
			const c = startCol + i;
			if (c < cols) {
				grid[0][c] = !grid[0][c]; // Flip in first row for simplicity
			}
		}

		message = `Flipped word: ${word}`;
		currentWordIndex++;
	}

	/**
	 * 30-cell storm: Randomly flip 30 cells
	 */
	function storm30() {
		let count = 0;
		while (count < 30) {
			const r = Math.floor(Math.random() * rows);
			const c = Math.floor(Math.random() * cols);
			if (!grid[r][c]) {
				grid[r][c] = true;
				count++;
			}
		}
		message = '30-cell storm complete.';
	}

	/**
	 * Row-shift sound: Simulate a sound effect by toggling a row
	 */
	function shiftRowSound() {
		const row = Math.floor(Math.random() * rows);
		for (let c = 0; c < cols; c++) {
			grid[row][c] = !grid[row][c];
		}
		message = `Shifted row ${row} (sound effect simulated).`;
	}

	/**
	 * Reset grid
	 */
	function resetGrid() {
		grid = Array.from({ length: rows }, () => Array(cols).fill(false));
		message = 'Grid reset.';
	}
</script>
