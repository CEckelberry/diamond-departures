+"""Status endpoints: season-state and freshness."""
+from fastapi import APIRouter, Response
+from apps.api.src.reader import SeasonStateReader, FreshnessReader
+
+router = APIRouter()
+
+# Initialize readers (singleton-like for this scope)
+_season_reader = SeasonStateReader()
+_freshness_reader = FreshnessReader()
+
+
+@router.get("/season-state", tags=["status"])
+def get_season_state(response: Response):
+    """
+    Get the current season state.
+    Cached for 5 minutes (max-age=300).
+    """
+    season = _season_reader.get_current_season()
+    response.headers["Cache-Control"] = "max-age=300"
+    return {
+        "season_id": season.season_id,
+        "start_date": season.start_date,
+        "end_date": season.end_date,
+        "is_active": season.is_active
+    }
+
+
+@router.get("/freshness", tags=["status"])
+def get_freshness(response: Response):
+    """
+    Get the system freshness status.
+    Not cached (no-store).
+    """
+    freshness = _freshness_reader.get_freshness_status()
+    response.headers["Cache-Control"] = "no-store"
+    return {
+        "last_updated": freshness.last_updated,
+        "age_seconds": freshness.age_seconds,
+        "is_fresh": freshness.is_fresh
+    }
