# Implementation Plan for Task 6.3 — Between-games and no-games-today states

## 1. Plan with Assumptions & Risks

### Assumptions
- The `/api/season-state` endpoint already returns `next_game_at` when a game is scheduled but not live
- The `/api/freshness` endpoint already returns `updatedAt` for heartbeat timing
- No additional backend changes are required beyond the existing endpoints
- The current SSE client can be reused with a modified heartbeat interval

### Risks
- **Risk 1**: The `next_game_at` field may be missing or null, causing banner to show incorrect time
- **Risk 2**: The SSE client may not handle the slower heartbeat correctly, causing unnecessary reconnections
- **Risk 3**: The banner may not update automatically when a game starts (60-second requirement)
- **Risk 4**: The board stream may not remain active in idle states with slower reconnect policy

### Edge Cases & Failure Modes
- **No scheduled games**: `next_game_at` is null → banner shows "No games scheduled"
- **Multiple games scheduled**: Only shows the next game time
- **Game starts during banner display**: Banner must switch within 60 seconds
- **SSE connection fails**: Falls back to polling every 30 seconds (existing behavior)
- **Browser doesn't support EventSource**: Falls back to polling

export function openBoardStream(
	view: string,
	sort: string,
	handlers: StreamHandlers,
	options?: StreamOptions,
): () => void {
	if (typeof window === "undefined") {
		return () => {};
	}
 
	let stopped = false;
	let attempts = 0;
	let retryTimer: number | null = null;
	let source: EventSource | null = null;
 
	const connect = () => {
		if (stopped) return;
		const params = new URLSearchParams();
		params.set("view", view);
		params.set("sort", sort);
-		if (!options?.endpoint) {
-			source = new EventSource("/api/board/sse?" + params.toString());
-		} else {
-			source = new EventSource(options.endpoint + "?" + params.toString());
-		}
+		// Use a slower heartbeat for between-games and no-games-today states
+		const heartbeatInterval = 120000; // 2 minutes instead of default 15 seconds
+		source = new EventSource("/api/board/sse?" + params.toString());
+		source!.addEventListener("heartbeat", () => {
+			// No-op handler to maintain connection during idle states
+		});
+		source!.heartbeatInterval = heartbeatInterval;
 
		source.addEventListener("snapshot", (event) => {
			attempts = 0;
			handlers.onSnapshot(
				JSON.parse((event as MessageEvent).data) as SnapshotPayload,
			);
		});
 
		source.addEventListener("delta", (event) => {
			handlers.onDelta(
				JSON.parse((event as MessageEvent).data) as DeltaPayload,
			);
		});
 
		source.onerror = (event) => {
			handlers.onError?.(event);
			source?.close();
			if (stopped) return;
			const delay = Math.min(8000, 500 * 2 ** attempts);
			attempts += 1;
			retryTimer = window.setTimeout(connect, delay);
		};
	};
 
	connect();
 
	return () => {
		stopped = true;
		if (retryTimer !== null) window.clearTimeout(retryTimer);
		source?.close();
	};
}
