"""
Leaderboard materialization and impacted view mapping.
"""

from typing import Set, Tuple, List
from django.db import transaction
from apps.ingest.models import PlayerStat, LeaderboardView


def get_impacted_views(
    updated_stat_names: List[str],
    updated_position_eligibilities: List[Tuple[int, bool]]  # (player_id, is_eligible)
) -> Set[Tuple[str, str]]:
    """
    Determine which (view_key, sort_stat) combos are impacted by updates.
    
    A view is impacted if:
    1. Its sort_stat is in the updated_stat_names.
    2. Any player whose position eligibility changed is eligible for that view's position filter (if any).
       For simplicity, we assume views are not position-filtered in this basic implementation,
       or that position eligibility changes always impact all views if the player was previously
       ineligible and is now eligible, or vice versa.
    
    For this packet, we simplify: if any stat name is updated, all views sorting by that stat are impacted.
    If position eligibility changes, we assume all views are potentially impacted because eligibility
    might affect who can be in the leaderboard.
    """
    impacted = set()
    
    # Get all existing leaderboard views
    all_views = LeaderboardView.objects.all()
    
    for view in all_views:
        # Check if the sort_stat is among the updated stats
        if view.sort_stat in updated_stat_names:
            impacted.add((view.view_key, view.sort_stat))
            continue
        
        # If position eligibility changed, we assume the view might be impacted.
        # In a more complex system, we would check if the player's position is relevant to the view.
        if updated_position_eligibilities:
            impacted.add((view.view_key, view.sort_stat))
            
    return impacted


@transaction.atomic
def materialize_leaderboard(view_key: str, sort_stat: str, top_n: int = 100) -> None:
    """
    Rebuild the top-N leaderboard for a given view_key and sort_stat.
    
    Deletes existing entries for this combo and inserts the new top-N.
    """
    # Delete existing entries for this view and stat
    from apps.ingest.models import LeaderboardEntry
    
    LeaderboardEntry.objects.filter(
        view_key=view_key,
        sort_stat=sort_stat
    ).delete()
    
    # Query the top N players sorted by the stat descending
    # Assuming PlayerStat has a value field and we want the latest value per player
    # This is a simplified query. In reality, you might need to aggregate or join.
    # For this packet, we assume PlayerStat.value is the current value for the stat.
    
    # We need to get the latest value for each player for the given stat_name (sort_stat)
    # This requires a subquery or annotation.
    
    from django.db.models import Max
    
    # Get the latest stat value for each player for the specific stat_name
    latest_stats = PlayerStat.objects.filter(
        stat_name=sort_stat
    ).values('player_id').annotate(
        latest_value=Max('value')
    ).order_by('-latest_value')[:top_n]
    
    # Create new leaderboard entries
    entries_to_insert = []
    for rank, stat_entry in enumerate(latest_stats, start=1):
        entries_to_insert.append(
            LeaderboardEntry(
                view_key=view_key,
                sort_stat=sort_stat,
                player_id=stat_entry['player_id'],
                value=stat_entry['latest_value'],
                rank=rank
            )
        )
    
    LeaderboardEntry.objects.bulk_create(entries_to_insert)

import pytest
from django.test import TestCase
from apps.ingest.models import PlayerStat, LeaderboardView, LeaderboardEntry
from apps.ingest.services.leaderboard import get_impacted_views, materialize_leaderboard


class TestImpactedViews(TestCase):
    def setUp(self):
        # Create some leaderboard views
        LeaderboardView.objects.create(view_key='overall', sort_stat='points')
        LeaderboardView.objects.create(view_key='assists', sort_stat='assists')
        LeaderboardView.objects.create(view_key='rebounds', sort_stat='rebounds')
        
    def test_impacted_by_stat_name(self):
        # Updating 'points' should impact 'overall' view
        impacted = get_impacted_views(['points'], [])
        self.assertIn(('overall', 'points'), impacted)
        self.assertNotIn(('assists', 'assists'), impacted)
        
    def test_impacted_by_position_eligibility(self):
        # Updating position eligibility should impact all views
        impacted = get_impacted_views([], [(1, True)])
        self.assertEqual(len(impacted), 3)  # All 3 views
        
    def test_no_impact(self):
        # No updates
        impacted = get_impacted_views([], [])
        self.assertEqual(len(impacted), 0)


class TestMaterializeLeaderboard(TestCase):
    def setUp(self):
        # Create a leaderboard view
        self.view = LeaderboardView.objects.create(view_key='test', sort_stat='points')
        
        # Create some players and stats
        from apps.ingest.models import Player
        
        self.player1 = Player.objects.create(name='Player 1')
        self.player2 = Player.objects.create(name='Player 2')
        self.player3 = Player.objects.create(name='Player 3')
        
        # Create stats with different values
        PlayerStat.objects.create(player=self.player1, stat_name='points', value=100.0)
        PlayerStat.objects.create(player=self.player2, stat_name='points', value=200.0)
        PlayerStat.objects.create(player=self.player3, stat_name='points', value=150.0)
        
    def test_materialize_top_3(self):
        # Materialize top 3
        materialize_leaderboard('test', 'points', top_n=3)
        
        # Check entries
        entries = LeaderboardEntry.objects.filter(view_key='test', sort_stat='points').order_by('rank')
        
        self.assertEqual(entries.count(), 3)
        
        # Check ranks and values
        self.assertEqual(entries[0].player, self.player2)  # 200 points
        self.assertEqual(entries[0].rank, 1)
        self.assertEqual(entries[0].value, 200.0)
        
        self.assertEqual(entries[1].player, self.player3)  # 150 points
        self.assertEqual(entries[1].rank, 2)
        self.assertEqual(entries[1].value, 150.0)
        
        self.assertEqual(entries[2].player, self.player1)  # 100 points
        self.assertEqual(entries[2].rank, 3)
        self.assertEqual(entries[2].value, 100.0)
        
    def test_materialize_top_2(self):
        # Materialize top 2
        materialize_leaderboard('test', 'points', top_n=2)
        
        entries = LeaderboardEntry.objects.filter(view_key='test', sort_stat='points').order_by('rank')
        
        self.assertEqual(entries.count(), 2)
        
        self.assertEqual(entries[0].player, self.player2)
        self.assertEqual(entries[0].rank, 1)
        
        self.assertEqual(entries[1].player, self.player3)
        self.assertEqual(entries[1].rank, 2)
        
    def test_delete_insert_semantics(self):
        # First materialize top 3
        materialize_leaderboard('test', 'points', top_n=3)
        
        # Update player1's points to 300
        PlayerStat.objects.filter(player=self.player1, stat_name='points').update(value=300.0)
        
        # Materialize again
        materialize_leaderboard('test', 'points', top_n=3)
        
        entries = LeaderboardEntry.objects.filter(view_key='test', sort_stat='points').order_by('rank')
        
        self.assertEqual(entries.count(), 3)
        
        # Player1 should now be first
        self.assertEqual(entries[0].player, self.player1)
        self.assertEqual(entries[0].rank, 1)
        self.assertEqual(entries[0].value, 300.0)
        
        self.assertEqual(entries[1].player, self.player2)
        self.assertEqual(entries[1].rank, 2)
        
        self.assertEqual(entries[2].player, self.player3)
        self.assertEqual(entries[2].rank, 3)
