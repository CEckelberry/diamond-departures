```typescript
export interface AnimationOptions {
  duration?: number;
  reducedMotion?: boolean;
}

export class FlapAnimationSequencer {
  private queue: Promise<void>[] = [];
  private isAnimating = false;
  private reducedMotion: boolean;

  constructor(options: AnimationOptions = {}) {
    this.reducedMotion = options.reducedMotion ?? this.checkReducedMotion();
  }

  private checkReducedMotion(): boolean {
    if (typeof window === 'undefined') return false;
    return window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  }

  /**
   * Enqueue an animation task.
   * If already animating, this waits for the previous task to complete
   * before starting, ensuring only the latest state is rendered.
   */
  async animate(
    onAnimate: () => Promise<void>,
    options: AnimationOptions = {}
  ): Promise<void> {
    const { duration = 450 } = options;
    
    // If reduced motion is preferred, skip complex animation
    if (this.reducedMotion) {
      return this.animateReducedMotion(onAnimate);
    }

    const task = (async () => {
      // Wait for any previous tasks in the queue
      const prev = this.queue[this.queue.length - 1];
      if (prev) {
        await prev;
      }
      
      this.isAnimating = true;
      try {
        await onAnimate();
      } finally {
        this.isAnimating = false;
      }
    })();

    this.queue.push(task);
    
    // Clean up completed tasks from the queue array to prevent memory leaks
    const cleanup = () => {
      const index = this.queue.indexOf(task);
      if (index > -1) {
        this.queue.splice(index, 1);
      }
    };
    
    task.then(cleanup).catch(cleanup);
  }

  private async animateReducedMotion(onAnimate: () => Promise<void>): Promise<void> {
    // Instant swap
    await onAnimate();
    
    // Optional: Trigger a flash effect if the caller supports it via a callback
    // For now, we just ensure the value is updated instantly.
    // A 200ms flash can be implemented by the component using a temporary class.
  }

  /**
   * Check current reduced motion preference.
   */
  isReducedMotion(): boolean {
    return this.reducedMotion;
  }
}

// Singleton instance for shared state across cells if needed, 
// or instantiate per-cell. For single-cell flap, per-cell is safer for isolation.
export const createSequencer = (options?: AnimationOptions) => new FlapAnimationSequencer(options);
```

<script lang="ts">
  import { createSequencer, type AnimationOptions } from './animation';
  import { onMount } from 'svelte';

  export let value: string = '';
  export let previousValue: string = '';
  
  // Animation configuration
  const PHASE_1_DURATION = 150; // Flip up
  const PHASE_2_DURATION = 150; // Swap (invisible)
  const PHASE_3_DURATION = 150; // Flip down
  const TOTAL_DURATION = PHASE_1_DURATION + PHASE_2_DURATION + PHASE_3_DURATION;

  let sequencer: ReturnType<typeof createSequencer>;
  let currentDisplay: string = '';
  let nextDisplay: string = '';
  let isFlippingUp = false;
  let isFlippingDown = false;
  let isSwapping = false;
  let flashClass = '';

  onMount(() => {
    sequencer = createSequencer();
    currentDisplay = value;
    nextDisplay = value;
  });

  // Watch for value changes
  $: if (value !== currentDisplay) {
    previousValue = currentDisplay;
    animateChange(value);
  }

  async function animateChange(newValue: string) {
    if (sequencer.isReducedMotion()) {
      handleReducedMotion(newValue);
      return;
    }

    await sequencer.animate(async () => {
      // Phase 1: Flip Up (Current digit rotates out)
      isFlippingUp = true;
      isFlippingDown = false;
      isSwapping = false;
      
      await wait(PHASE_1_DURATION);

      // Phase 2: Swap Content (Invisible transition)
      isFlippingUp = false;
      isSwapping = true;
      currentDisplay = newValue; // Swap the underlying value
      
      await wait(PHASE_2_DURATION);

      // Phase 3: Flip Down (New digit rotates in)
      isSwapping = false;
      isFlippingDown = true;
      
      await wait(PHASE_3_DURATION);

      // Reset state
      isFlippingDown = false;
      currentDisplay = newValue;
    }, { duration: TOTAL_DURATION });
  }

  function handleReducedMotion(newValue: string) {
    // Instant swap
    currentDisplay = newValue;
    
    // 200ms flash effect
    flashClass = 'flap-flash';
    setTimeout(() => {
      flashClass = '';
    }, 200);
  }

  function wait(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
</script>
