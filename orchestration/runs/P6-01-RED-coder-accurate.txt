// src/components/Header/OffSeasonBanner.test.tsx

import React from 'react';
import { render, screen } from '@testing-library/react';
import Header from './Header'; // Adjust path as necessary
import { AppContext } from '../../context/AppContext';

describe('OffSeasonBanner', () => {
  const mockSetMode = jest.fn();

  const renderWithOffSeason = (ui: React.ReactElement) => {
    return render(
      <AppContext.Provider value={{ mode: 'off-season', setMode: mockSetMode }}>
        {ui}
      </AppContext.Provider>
    );
  };

  it('should show off-season banner copy when mode is off-season', () => {
    renderWithOffSeason(<Header />);
    
    // Assert banner text presence
    expect(screen.getByText(/Off Season/i)).toBeInTheDocument();
    expect(screen.getByText(/Season ends in/i)).toBeInTheDocument();
  });

  it('should render countdown hook when mode is off-season', () => {
    renderWithOffSeason(<Header />);
    
    // Assert countdown component/hook output is present
    // Adjust selector based on actual implementation (e.g., data-testid or specific text)
    expect(screen.getByTestId('countdown-hook')).toBeInTheDocument();
  });

  it('should NOT show off-season banner when mode is active', () => {
    render(
      <AppContext.Provider value={{ mode: 'active', setMode: mockSetMode }}>
        <Header />
      </AppContext.Provider>
    );
    
    expect(screen.queryByText(/Off Season/i)).not.toBeInTheDocument();
    expect(screen.queryByTestId('countdown-hook')).not.toBeInTheDocument();
  });
});
