# Packet P0-02 — Database schema + migrations

## 1. Concise Plan

1.  **Schema Design (`001_init.sql`)**: Create tables for `teams`, `players`, `games`, `game_events`, and `stats`. Use `numeric(8,3)` for all decimal stats. Add necessary indexes (PKs, FKs, and query-specific indexes like `game_date` and `team_id`). Ensure idempotency using `IF NOT EXISTS` where appropriate or standard migration tool conventions (assuming `golang-migrate` or similar based on `Makefile` context).
2.  **Seed Data (`002_seed_teams.sql`)**: Insert the 30 MLB teams into the `teams` table.
3.  **Park Factors (`003_park_factors.sql`)**: Create the `park_factors` table and insert 2026 data.
4.  **Content Files**: Generate `teams.json` and `park-factors-2026.json` with realistic/mock data consistent with the schema.
5.  **Makefile**: Add `migrate` target using `DATABASE_URL`.

### `apps/api/migrations/001_init.sql`

```sql
-- +migrate Up
-- +migrate StatementBegin
CREATE TABLE IF NOT EXISTS teams (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    abbreviation VARCHAR(5) NOT NULL UNIQUE,
    city VARCHAR(50) NOT NULL,
    league VARCHAR(10) NOT NULL CHECK (league IN ('AL', 'NL')),
    division VARCHAR(20) NOT NULL,
    stadium_name VARCHAR(100),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS players (
    id SERIAL PRIMARY KEY,
    team_id INTEGER NOT NULL REFERENCES teams(id) ON DELETE CASCADE,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    position VARCHAR(10) NOT NULL CHECK (position IN ('P', 'C', '1B', '2B', '3B', 'SS', 'LF', 'CF', 'RF', 'DH')),
    jersey_number INTEGER,
    birth_date DATE,
    bats VARCHAR(5) CHECK (bats IN ('L', 'R', 'S')),
    throws VARCHAR(5) CHECK (throws IN ('L', 'R', 'S')),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS games (
    id SERIAL PRIMARY KEY,
    game_date DATE NOT NULL,
    home_team_id INTEGER NOT NULL REFERENCES teams(id),
    away_team_id INTEGER NOT NULL REFERENCES teams(id),
    home_score INTEGER,
    away_score INTEGER,
    status VARCHAR(20) DEFAULT 'scheduled', -- scheduled, in_progress, final, cancelled
    venue_id INTEGER, -- Optional link to stadium
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS game_events (
    id SERIAL PRIMARY KEY,
    game_id INTEGER NOT NULL REFERENCES games(id) ON DELETE CASCADE,
    inning INTEGER NOT NULL,
    half_inning VARCHAR(5) CHECK (half_inning IN ('top', 'bottom')),
    event_type VARCHAR(50) NOT NULL, -- single, double, triple, home_run, walk, strikeout, etc.
    batter_id INTEGER REFERENCES players(id),
    pitcher_id INTEGER REFERENCES players(id),
    fielder_id INTEGER REFERENCES players(id), -- Primary fielder involved
    description TEXT,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS stats (
    id SERIAL PRIMARY KEY,
    player_id INTEGER NOT NULL REFERENCES players(id) ON DELETE CASCADE,
    game_id INTEGER REFERENCES games(id) ON DELETE SET NULL,
    season_year INTEGER NOT NULL,
    at_bats INTEGER DEFAULT 0,
    runs INTEGER DEFAULT 0,
    hits INTEGER DEFAULT 0,
    doubles INTEGER DEFAULT 0,
    triples INTEGER DEFAULT 0,
    home_runs INTEGER DEFAULT 0,
    rbi INTEGER DEFAULT 0,
    stolen_bases INTEGER DEFAULT 0,
    caught_stealing INTEGER DEFAULT 0,
    walks INTEGER DEFAULT 0,
    strikeouts INTEGER DEFAULT 0,
    batting_avg NUMERIC(8,3),
    on_base_pct NUMERIC(8,3),
    slugging_pct NUMERIC(8,3),
    ops NUMERIC(8,3),
    earned_runs INTEGER DEFAULT 0,
    innings_pitched NUMERIC(8,3),
    hits_allowed INTEGER DEFAULT 0,
    walks_allowed INTEGER DEFAULT 0,
    strikeouts_pitched INTEGER DEFAULT 0,
    era NUMERIC(8,3),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS park_factors (
    id SERIAL PRIMARY KEY,
    team_id INTEGER NOT NULL REFERENCES teams(id) ON DELETE CASCADE,
    season_year INTEGER NOT NULL,
    factor_offensive NUMERIC(8,3), -- >1.0 is hitter-friendly
    factor_pitching NUMERIC(8,3), -- <1.0 is pitcher-friendly
    factor_run NUMERIC(8,3),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    UNIQUE(team_id, season_year)
);

-- Indexes
CREATE INDEX IF NOT EXISTS idx_players_team_id ON players(team_id);
CREATE INDEX IF NOT EXISTS idx_games_date ON games(game_date);
CREATE INDEX IF NOT EXISTS idx_games_home_team ON games(home_team_id);
CREATE INDEX IF NOT EXISTS idx_games_away_team ON games(away_team_id);
CREATE INDEX IF NOT EXISTS idx_game_events_game_id ON game_events(game_id);
CREATE INDEX IF NOT EXISTS idx_stats_player_id ON stats(player_id);
CREATE INDEX IF NOT EXISTS idx_stats_season ON stats(season_year);
CREATE INDEX IF NOT EXISTS idx_park_factors_team_id ON park_factors(team_id);
CREATE INDEX IF NOT EXISTS idx_park_factors_season ON park_factors(season_year);

-- +migrate StatementEnd

-- +migrate Down
-- +migrate StatementBegin
DROP TABLE IF EXISTS park_factors;
DROP TABLE IF EXISTS stats;
DROP TABLE IF EXISTS game_events;
DROP TABLE IF EXISTS games;
DROP TABLE IF EXISTS players;
DROP TABLE IF EXISTS teams;
-- +migrate StatementEnd
```
