import React, { createContext, useContext, useEffect, useState } from 'react';

type PreviewMode = 'live' | 'preview' | 'none';
interface PreviewModeContextType {
  mode: PreviewMode;
  setMode: (mode: PreviewMode) => void;
  isPreviewMode: boolean;
}

const PreviewModeContext = createContext<PreviewModeContextType | undefined>(undefined);

export const usePreviewMode = () => {
  const context = useContext(PreviewModeContext);
  if (!context) {
    throw new Error('usePreviewMode must be used within PreviewModeProvider');
  }
  return context;
};

export const PreviewModeProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [mode, setMode] = useState<PreviewMode>('none');

  return (
    <PreviewModeContext.Provider value={{ mode, setMode, isPreviewMode: mode === 'preview' }}>
      {children}
    </PreviewModeContext.Provider>
  );
};

### 2. Create SSE Service for Preview Mode

```typescript
// apps/web/src/services/previewSSEService.ts
interface SSEMessage {
  type: string;
  data: any;
  id?: string;
  retry?: number;
}

export class PreviewSSEService {
  private eventSource: EventSource | null = null;
  private onMessageCallbacks: ((message: SSEMessage) => void)[] = [];
  private reconnectInterval: NodeJS.Timeout | null = null;
  private shouldReconnect = true;

  constructor(private previewMode: boolean) {
    this.previewMode = previewMode;
  }

  public start(): Promise<void> {
    return new Promise((resolve, reject) => {
      if (this.previewMode) {
        this.eventSource = new EventSource('/api/board/preview-sse');
      } else {
        this.eventSource = new EventSource('/api/board/sse');
      }

      this.eventSource.onopen = () => {
        console.log('SSE connection opened');
        this.shouldReconnect = true;
        resolve();
      };

      this.eventSource.onmessage = (event) => {
        try {
          const message: SSEMessage = JSON.parse(event.data);
          this.onMessageCallbacks.forEach(callback => callback(message));
        } catch (error) {
          console.error('Error parsing SSE message:', error);
        }
      };

      this.eventSource.onerror = (error) => {
        console.error('SSE connection error:', error);
        this.shouldReconnect = false;
        reject(error);
      };

      this.eventSource.onclose = () => {
        console.log('SSE connection closed');
        if (this.shouldReconnect) {
          this.reconnectInterval = setTimeout(() => {
            this.start().catch(console.error);
          }, 5000);
        }
      };
    });
  }

  public stop(): void {
    this.shouldReconnect = false;
    if (this.reconnectInterval) {
      clearTimeout(this.reconnectInterval);
    }
    if (this.eventSource) {
      this.eventSource.close();
      this.eventSource = null;
    }
  }

  public onMessage(callback: (message: SSEMessage) => void): () => void {
    this.onMessageCallbacks.push(callback);
    return () => {
      const index = this.onMessageCallbacks.indexOf(callback);
      if (index > -1) {
        this.onMessageCallbacks.splice(index, 1);
      }
    };
  }
}
```

### 5. Create SSE Provider with Preview Support

```typescript
// apps/web/src/providers/SSEProvider.tsx
import React, { useEffect, useRef } from 'react';
import { usePreviewMode } from '../contexts/PreviewModeContext';
import { PreviewSSEService } from '../services/previewSSEService';

interface SSEProviderProps {
  children: React.ReactNode;
}

export const SSEProvider: React.FC<SSEProviderProps> = ({ children }) => {
  const { mode, isPreviewMode } = usePreviewMode();
  const serviceRef = useRef<PreviewSSEService | null>(null);

  useEffect(() => {
    if (serviceRef.current) {
      serviceRef.current.stop();
    }

    if (isPreviewMode) {
      serviceRef.current = new PreviewSSEService(true);
    } else {
      serviceRef.current = new PreviewSSEService(false);
    }

    serviceRef.current.start().catch(console.error);

    return () => {
      if (serviceRef.current) {
        serviceRef.current.stop();
      }
    };
  }, [isPreviewMode]);

  return <>{children}</>;
};
```

    await page.click('button:has-text("Exit Preview Mode")');
    await expect(page.locator('text=Enter Preview Mode')).toBeVisible();
    await expect(page.locator('text=Preview Mode Active')).not.toBeVisible();

    await page.goto('/board');
