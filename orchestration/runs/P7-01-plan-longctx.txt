   private startAnimation(): void {

         this.animationId = this.rafManager.requestAnimationFrame(animate);
         return;

       this.updateAnimations();
       this.animationId = this.rafManager.requestAnimationFrame(animate);

       this.rafManager.cancelAnimationFrame(this.animationId);
       this.animationId = null;
     }
   }
 
   private updateAnimations(): void {
     // Optimized animation updates with batching
     const cellsToAnimate = this.cells.filter(cell => {
       const rect = cell.getBoundingClientRect();
       return rect.width > 0 && rect.height > 0; // Check if visible
     });
 
     cellsToAnimate.forEach(cell => {
       // Batch DOM reads
       const transform = cell.dataset.transform || '';
       const opacity = cell.dataset.opacity || '';
       
       // Batch DOM writes
       requestAnimationFrame(() => {
         cell.style.transform = transform;
         cell.style.opacity = opacity;
       });
     });
   }

     this.cells.push(cell);
   }
 
   public removeCell(cell: HTMLElement): void {
     const index = this.cells.indexOf(cell);

export const useReducedMotion = () => {
   const [prefersReducedMotion, setPrefersReducedMotion] = useState(false);
 
   useEffect(() => {
     const mediaQuery = window.matchMedia('(prefers-reduced-motion: reduce)');
     setPrefersReducedMotion(mediaQuery.matches);
 
     const handleChange = (e: MediaQueryListEvent) => {
       setPrefersReducedMotion(e.matches);
     };
 
     mediaQuery.addEventListener('change', handleChange);
     return () => mediaQuery.removeEventListener('change', handleChange);
   }, []);
 
   return prefersReducedMotion;

export class FlapAnimation {
  private element: HTMLElement;
  private animationId: number | null = null;
 
  constructor(element: HTMLElement) {
    this.element = element;
  }
 
  public destroy(): void {
    if (this.animationId !== null) {
       cancelAnimationFrame(this.animationId);
       this.animationId = null;
    }
    this.element.style.transform = '';
    this.element.style.transition = '';
  }
 
  public animate(): void {
     // Optimized animation with frame rate limiting
     const startTime = performance.now();
     const duration = 300; // ms
 
     const animate = (currentTime: number) => {
       const elapsed = currentTime - startTime;
       const progress = Math.min(elapsed / duration, 1);
 
       // Apply easing
       const easeProgress = this.easeInOutCubic(progress);
       this.element.style.transform = `rotateY(${easeProgress * 180}deg)`;
 
       if (progress < 1) {
         this.animationId = requestAnimationFrame(animate);
       }
     };
 
     this.animationId = requestAnimationFrame(animate);
   }
 
  private easeInOutCubic(t: number): number {
     return t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
  }
 }
