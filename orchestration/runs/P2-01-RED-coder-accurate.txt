```python
import os
import pytest
from unittest.mock import patch

# Assuming the implementation will be in apps.api.config
# We import here to ensure the module structure is recognized, 
# but the test will fail if the module doesn't exist or logic is wrong.
from apps.api.config import get_settings


class TestConfigLoader:
    """Tests for configuration loading from environment variables."""

    @pytest.fixture(autouse=True)
    def clear_env(self):
        """Clear relevant env vars before each test to ensure isolation."""
        with patch.dict(os.environ, {}, clear=True):
            yield

    def test_loads_port_from_env(self):
        """Test that PORT is read from environment variables."""
        with patch.dict(os.environ, {"PORT": "8000"}):
            settings = get_settings()
            assert settings.PORT == 8000

    def test_loads_log_level_from_env(self):
        """Test that LOG_LEVEL is read from environment variables."""
        with patch.dict(os.environ, {"LOG_LEVEL": "DEBUG"}):
            settings = get_settings()
            assert settings.LOG_LEVEL == "DEBUG"

    def test_loads_database_url_from_env(self):
        """Test that DATABASE_URL is read from environment variables."""
        db_url = "postgresql://user:pass@localhost:5432/dbname"
        with patch.dict(os.environ, {"DATABASE_URL": db_url}):
            settings = get_settings()
            assert settings.DATABASE_URL == db_url

    def test_defaults_when_env_not_set(self):
        """Test that default values are used when env vars are missing."""
        # This test ensures we have sensible defaults if env is empty
        settings = get_settings()
        # Assuming default port is 8000, log level INFO, and a dummy DB URL
        assert settings.PORT == 8000
        assert settings.LOG_LEVEL == "INFO"
        assert settings.DATABASE_URL == "sqlite:///./test.db"
```

**File: `apps/api/tests/test_health.py`**

```python
import pytest
from unittest.mock import AsyncMock, patch
from httpx import AsyncClient, ASGITransport
from apps.api.main import create_app


class TestHealthEndpoint:
    """Tests for the /api/health endpoint behavior."""

    @pytest.fixture
    def app(self):
        """Create a FastAPI application instance."""
        return create_app()

    @pytest.mark.asyncio
    async def test_health_returns_ok_when_db_healthy(self, app):
        """Test that health returns 'ok' status when DB checker succeeds."""
        transport = ASGITransport(app=app)
        async with AsyncClient(transport=transport, base_url="http://test") as client:
            # Mock the database checker to return True (healthy)
            with patch("apps.api.routers.health.check_database_connection", new_callable=AsyncMock) as mock_db:
                mock_db.return_value = True
                
                response = await client.get("/api/health")
                
                assert response.status_code == 200
                data = response.json()
                assert data["status"] == "ok"
                assert "database" in data

    @pytest.mark.asyncio
    async def test_health_returns_degraded_when_db_unhealthy(self, app):
        """Test that health returns 'degraded' status and HTTP 503 when DB checker fails."""
        transport = ASGITransport(app=app)
        async with AsyncClient(transport=transport, base_url="http://test") as client:
            # Mock the database checker to return False (unhealthy)
            with patch("apps.api.routers.health.check_database_connection", new_callable=AsyncMock) as mock_db:
                mock_db.return_value = False
                
                response = await client.get("/api/health")
                
                assert response.status_code == 503
                data = response.json()
                assert data["status"] == "degraded"
                assert "database" in data

    @pytest.mark.asyncio
    async def test_health_endpoint_exists(self, app):
        """Test that the /api/health route is mounted and accessible."""
        transport = ASGITransport(app=app)
        async with AsyncClient(transport=transport, base_url="http://test") as client:
            # Even if it fails due to missing implementation, this verifies routing structure
            # We expect a 500 or specific error if the route doesn't exist, 
            # but for RED, we just want to ensure the test framework hits the right path.
            # If the router isn't mounted, httpx will raise an error or return 404.
            try:
                response = await client.get("/api/health")
                # If we get here, the route exists. 
                # The specific assertions in other tests will handle the logic failure.
                assert response.status_code in [200, 503, 500] # Allow 500 for missing impl
            except Exception as e:
                # If the router isn't mounted at all, httpx might not raise, 
                # but FastAPI will return 404.
                assert False, f"Route /api/health not found or connection error: {e}"
```
