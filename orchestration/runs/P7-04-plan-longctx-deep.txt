2. **Missing metadata** should default to sensible values
3. **Image URLs** must be absolute and properly formatted
4. **Robots.txt** should handle crawling rules for different environments
5. **Sitemap.xml** must include all routes and handle pagination

import { PageMetadata } from '@/types/metadata';

interface MetadataContextType {
  metadata: PageMetadata;
  setMetadata: (metadata: PageMetadata) => void;
  ogImageRoute: string;
}

const MetadataContext = createContext<MetadataContextType | undefined>(undefined);

export const MetadataProvider: React.FC<{ children: ReactNode; initialMetadata?: PageMetadata }> = ({
  children,
  initialMetadata,
}) => {
  const [metadata, setMetadata] = React.useState<PageMetadata>(initialMetadata || {});
  const ogImageRoute = '/api/og-image';

  return (
    <MetadataContext.Provider value={{ metadata, setMetadata, ogImageRoute }}>
      {children}
    </MetadataContext.Provider>
  );
};

export const useMetadata = () => {
  const context = useContext(MetadataContext);
  if (!context) {
    throw new Error('useMetadata must be used within MetadataProvider');
  }
  return context;
};

import React from 'react';
import Head from 'next/head';
import { useRouter } from 'next/router';
import { useMetadata } from '@/contexts/MetadataContext';
import { PageMetadata } from '@/types/metadata';

interface SEOProps {
  children?: React.ReactNode;
  metadata?: Partial<PageMetadata>;
}

export const SEO: React.FC<SEOProps> = ({ children, metadata: customMetadata }) => {
  const router = useRouter();
  const { metadata, setMetadata, ogImageRoute } = useMetadata();
  
  // Merge default metadata with custom metadata
  const finalMetadata: PageMetadata = {
    ...metadata,
    ...customMetadata,
    url: `${process.env.NEXT_PUBLIC_SITE_URL || 'http://localhost:3000'}${router.asPath}`,
  };

  // Generate Open Graph image URL
  const ogImageUrl = finalMetadata.image 
    ? finalMetadata.image 
    : `${process.env.NEXT_PUBLIC_SITE_URL || 'http://localhost:3000'}${ogImageRoute}?title=${encodeURIComponent(finalMetadata.title)}`;

  return (
    <>
      <Head>
        <title>{finalMetadata.title}</title>
        <meta name="description" content={finalMetadata.description} />
        <meta name="keywords" content={finalMetadata.keywords?.join(', ') || ''} />
        
        {/* Open Graph */}
        <meta property="og:title" content={finalMetadata.title} />
        <meta property="og:description" content={finalMetadata.description} />
        <meta property="og:type" content={finalMetadata.type || 'website'} />
        <meta property="og:url" content={finalMetadata.url} />
        {finalMetadata.image && <meta property="og:image" content={ogImageUrl} />}
        {finalMetadata.image && <meta property="og:image:alt" content={finalMetadata.image?.replace(/-/g, ' ')} />}
        
        {/* Twitter */}
        <meta name="twitter:card" content="summary_large_image" />
        <meta name="twitter:title" content={finalMetadata.title} />
        <meta name="twitter:description" content={finalMetadata.description} />
        {finalMetadata.image && <meta name="twitter:image" content={ogImageUrl} />}
        
        {/* Canonical URL */}
        <link rel="canonical" href={finalMetadata.url} />
        
        {/* Alternate URLs */}
        {router.asPath !== '/' && (
          <link rel="alternate" hrefLang="x-default" href={finalMetadata.url} />
        )}
      </Head>
      
      {children}
    </>
  );
};

import { OgImageGenerator } from '@/lib/og-image-generator';

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'GET') {
    return res.status(405).json({ error: 'Method not allowed' });
  }

  try {
    const { title = 'Default Title' } = req.query;
    
    const generator = new OgImageGenerator();
    const imageBuffer = await generator.generateOgImage(title as string);
    
    res.setHeader('Content-Type', 'image/png');
    res.setHeader('Cache-Control', 'public, max-age=3600'); // Cache for 1 hour
    res.setHeader('Surrogate-Key', 'og-image');
    
    imageBuffer.pipe(res);
  } catch (error) {
    console.error('OG Image generation error:', error);
    res.status(500).json({ error: 'Failed to generate OG image' });
  }
}

### 5. Create OG Image Generator

**File**: `src/lib/og-image-generator.ts`
```typescript
// New file
import { createCanvas, CanvasRenderingContext2D } from 'canvas';
import { OgImageOptions } from '@/types/metadata';

export class OgImageGenerator {
  private canvas: any;
  private ctx: CanvasRenderingContext2D;
  private width = 1200;
  private height = 630;

  constructor(options?: OgImageOptions) {
    this.canvas = createCanvas(this.width, this.height);
    this.ctx = this.canvas.getContext('2d');
    
    // Set default options
    const opts = {
      backgroundColor: '#ffffff',
      textColor: '#000000',
      fontSize: 48,
      fontFamily: 'Arial, sans-serif',
      ...options,
    };

    // Apply background
    this.ctx.fillStyle = opts.backgroundColor;
    this.ctx.fillRect(0, 0, this.width, this.height);
  }

  async generateOgImage(title: string): Promise<Buffer> {
    try {
      // Set text properties
      this.ctx.fillStyle = '#000000';
      this.ctx.font = `bold ${this.height * 0.1}px ${this.ctx.font}`;
      this.ctx.textAlign = 'center';
      this.ctx.textBaseline = 'middle';

      // Wrap text
      const words = title.split(' ');
      const lines: string[] = [];
      let currentLine = '';

      for (const word of words) {
        const testLine = currentLine + (currentLine ? ' ' : '') + word;
        const metrics = this.ctx.measureText(testLine);
        if (metrics.width > this.width * 0.8 && currentLine) {
          lines.push(currentLine);
          currentLine = word;
        } else {
          currentLine = testLine;
        }
      }
      lines.push(currentLine);

      // Draw text
      const lineHeight = this.height * 0.1;
      const startY = this.height / 2 - (lines.length * lineHeight) / 2;

      lines.forEach((line, index) => {
        this.ctx.fillText(
          line,
          this.width / 2,
          startY + index * lineHeight
        );
      });

      // Return PNG buffer
      return this.canvas.toBuffer('image/png');
    } catch (error) {
      console.error('OG Image generation failed:', error);
      throw new Error('Failed to generate OG image');
    }
  }
}
```

### 7. Create Sitemap Generator

**File**: `src/lib/sitemap-generator.ts`
```typescript
// New file
import { GetStaticPathsResult } from 'next';
import { getAllPaths } from '@/lib/content-fetcher';

export interface SitemapEntry {
  loc: string;
  lastmod?: string;
  changefreq?: 'daily' | 'weekly' | 'monthly' | 'yearly' | 'never';
  priority?: number;
}

export class SitemapGenerator {
  private baseUrl: string;

  constructor(baseUrl: string) {
    this.baseUrl = baseUrl;
  }

  async generateSitemap(): Promise<SitemapEntry[]> {
    // Get all static paths
    const paths = await getAllPaths();
    
    const entries: SitemapEntry[] = [];
    
    // Add homepage
    entries.push({
      loc: this.baseUrl,
      changefreq: 'daily',
      priority: 1.0,
    });

    // Add all paths with appropriate priorities
    paths.forEach((path) => {
      const url = `${this.baseUrl}${path}`;
      
      // Determine priority based on path depth
      const depth = path.split('/').filter(Boolean).length;
      const priority = Math.max(0.1, 1 - (depth * 0.1));
      
      entries.push({
        loc: url,
        changefreq: 'weekly',
        priority,
      });
    });

    return entries;
  }

  generateSitemapXml(entries: SitemapEntry[]): string {
    const urls = entries.map(entry => `
  <url>
    <loc>${entry.loc}</loc>
    ${entry.lastmod ? `<lastmod>${entry.lastmod}</lastmod>` : ''}
    ${entry.changefreq ? `<changefreq>${entry.changefreq}</changefreq>` : ''}
    ${entry.priority ? `<priority>${entry.priority}</priority>` : ''}
  </url>
`).join('');

    return `<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
${urls}
</urlset>`;
  }
}
```

import { NextApiRequest, NextApiResponse } from 'next';
import { SitemapGenerator } from '@/lib/sitemap-generator';

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  try {
    const baseUrl = process.env.NEXT_PUBLIC_SITE_URL || 'http://localhost:3000';
    const generator = new SitemapGenerator(baseUrl);
    const entries = await generator.generateSitemap();
    const xml = generator.generateSitemapXml(entries);

    res.setHeader('Content-Type', 'application/xml');
    res.setHeader('Cache-Control', 'public, max-age=3600'); // Cache for 1 hour
    res.setHeader('Surrogate-Key', 'sitemap');
    
    res.status(200).send(xml);
  } catch (error) {
    console.error('Sitemap generation error:', error);
    res.status(500).json({ error: 'Failed to generate sitemap' });
  }
}
