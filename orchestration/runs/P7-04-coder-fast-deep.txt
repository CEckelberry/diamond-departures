# Packet P7-04 — SEO + Open Graph GREEN

## 1. Plan

### Assumptions
1.  **Framework**: The project uses a standard Node.js/TypeScript stack with `express` (or compatible middleware) and `react` (or compatible SSR/SSG layer). I will implement a generic but robust structure that fits most modern web apps.
2.  **Static Assets**: `/public` or `static` directory exists for serving `robots.txt` and `sitemap.xml`.
3.  **Image Generation**: We will use a placeholder route that returns a generated SVG/Canvas image for OG tags, as full image generation (e.g., via Puppeteer/Sharp) is often too heavy for a "baseline" green pass. The contract will be defined clearly.
4.  **Metadata Structure**: A centralized `metadata` object or context will be used to inject `<head>` tags.

### Risks & Edge Cases
1.  **Sitemap URL Accuracy**: The sitemap must reflect the actual deployed base URL. Hardcoding `localhost` is a risk for production. *Mitigation*: Use `process.env.BASE_URL` or request origin.
2.  **OG Image Dimensions**: OG images must be at least 200x200px, preferably 1200x630px. The placeholder will generate a valid SVG that scales.
3.  **Robots.txt Caching**: Bots cache aggressively. *Mitigation*: Set appropriate headers (`Cache-Control: public, max-age=86400`).
4.  **Dynamic Routes**: If the app has dynamic routes (e.g., `/blog/:slug`), the sitemap generator needs to know about them. *Mitigation*: For this baseline, we will implement a static list of known routes and a hook for dynamic ones.

### Implementation Steps
1.  Create `src/utils/metadata.ts`: Centralized metadata configuration.
2.  Create `src/routes/seo.ts`: Express routes for `/robots.txt` and `/sitemap.xml`.
3.  Create `src/routes/og-image.ts`: Placeholder route for `/og-image.png` (returns SVG).
4.  Update `src/app.ts` (or equivalent entry): Wire up SEO routes and ensure metadata is injected into the main HTML template.
5.  Add tests in `tests/seo.test.ts`.

### File: `src/routes/seo.ts`
```typescript
// src/routes/seo.ts
import { Router, Request, Response } from 'express';
import { generateMetadata } from '../utils/metadata';

const router = Router();

// Robots.txt
router.get('/robots.txt', (req: Request, res: Response) => {
  const baseUrl = req.get('host') ? `https://${req.get('host')}` : 'http://localhost:3000';
  const robots = `
User-agent: *
Allow: /
Sitemap: ${baseUrl}/sitemap.xml
`.trim();

  res.set('Content-Type', 'text/plain');
  res.set('Cache-Control', 'public, max-age=86400');
  res.send(robots);
});

// Sitemap.xml
router.get('/sitemap.xml', (req: Request, res: Response) => {
  const baseUrl = req.get('host') ? `https://${req.get('host')}` : 'http://localhost:3000';
  
  // Static routes for baseline
  const staticRoutes = ['/', '/about', '/contact', '/blog'];
  
  const lastmod = new Date().toISOString().split('T')[0];
  
  const urlset = `<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
${staticRoutes.map(route => `
  <url>
    <loc>${baseUrl}${route}</loc>
    <lastmod>${lastmod}</lastmod>
    <changefreq>weekly</changefreq>
    <priority>0.8</priority>
  </url>`).join('')}
</urlset>`.trim();

  res.set('Content-Type', 'application/xml');
  res.set('Cache-Control', 'public, max-age=3600');
  res.send(urlset);
});

export default router;
```

### File: `src/routes/og-image.ts`
```typescript
// src/routes/og-image.ts
import { Router, Request, Response } from 'express';

const router = Router();

/**
 * Placeholder OG Image Route.
 * In production, this should generate a PNG/SVG dynamically based on query params.
 * For baseline GREEN, we return a valid SVG that meets OG spec (1200x630).
 */
router.get('/og-image', (req: Request, res: Response) => {
  const title = req.query.title || 'My App';
  const description = req.query.description || 'Default Description';

  // Simple SVG placeholder
  const svg = `
    <svg xmlns="http://www.w3.org/2000/svg" width="1200" height="630">
      <rect width="100%" height="100%" fill="#1a1a1a"/>
      <text x="50%" y="50%" dominant-baseline="middle" text-anchor="middle" fill="#ffffff" font-family="sans-serif" font-size="48">
        ${title}
      </text>
      <text x="50%" y="60%" dominant-baseline="middle" text-anchor="middle" fill="#cccccc" font-family="sans-serif" font-size="24">
        ${description}
      </text>
    </svg>
  `.trim();

  res.set('Content-Type', 'image/svg+xml');
  res.set('Cache-Control', 'public, max-age=86400');
  res.send(svg);
});

export default router;
```

### File: `tests/seo.test.ts`

```typescript
// tests/seo.test.ts
import request from 'supertest';
import app from '../src/app';

describe('SEO Routes', () => {
  describe('GET /api/robots.txt', () => {
    it('should return robots.txt with correct content type', async () => {
      const res = await request(app).get('/api/robots.txt');
      expect(res.status).toBe(200);
      expect(res.headers['content-type']).toContain('text/plain');
      expect(res.text).toContain('User-agent: *');
      expect(res.text).toContain('Sitemap:');
    });

    it('should include correct sitemap URL', async () => {
      const res = await request(app).get('/api/robots.txt');
      expect(res.text).toContain('Sitemap: http://localhost:3000/sitemap.xml');
    });
  });

  describe('GET /api/sitemap.xml', () => {
    it('should return sitemap.xml with correct content type', async () => {
      const res = await request(app).get('/api/sitemap.xml');
      expect(res.status).toBe(200);
      expect(res.headers['content-type']).toContain('application/xml');
      expect(res.text).toContain('<urlset');
    });

    it('should include static routes', async () => {
      const res = await request(app).get('/api/sitemap.xml');
      expect(res.text).toContain('<loc>http://localhost:3000/</loc>');
      expect(res.text).toContain('<loc>http://localhost:3000/about</loc>');
    });
  });

  describe('GET /api/og-image', () => {
    it('should return SVG with correct content type', async () => {
      const res = await request(app).get('/api/og-image');
      expect(res.status).toBe(200);
      expect(res.headers['content-type']).toContain('image/svg+xml');
      expect(res.text).toContain('<svg');
    });

    it('should include title in SVG', async () => {
      const res = await request(app).get('/api/og-image?title=Test+Title');
      expect(res.text).toContain('Test Title');
    });
  });
});
```
