<script>
  import { onMount, onDestroy } from 'svelte';
  import { createEventDispatcher } from 'svelte';

  export let item;
  export let index;
  export let onActivate;
  export let role = 'button';
  export let tabindex = 0;
  export let 'aria-label' = '';
  export let 'aria-selected' = false;
  export let 'aria-expanded' = false;
  export let 'aria-controls' = '';
  export let 'aria-describedby' = '';

  const dispatch = createEventDispatcher();

  let rowElement;
  let isFocused = false;

  function handleKeyDown(event) {
    switch (event.key) {
      case 'Enter':
      case ' ':
        event.preventDefault();
        if (onActivate) {
          onActivate(item, index);
        }
        dispatch('activate', { item, index });
        break;
      case 'ArrowDown':
        event.preventDefault();
        dispatch('navigate', { direction: 'down', item, index });
        break;
      case 'ArrowUp':
        event.preventDefault();
        dispatch('navigate', { direction: 'up', item, index });
        break;
      case 'Escape':
        event.preventDefault();
        dispatch('escape', { item, index });
        break;
    }
  }

  function handleClick() {
    if (onActivate) {
      onActivate(item, index);
    }
    dispatch('activate', { item, index });
  }

  function handleFocus() {
    isFocused = true;
    if (rowElement) {
      rowElement.setAttribute('aria-selected', 'true');
    }
  }

  function handleBlur() {
    isFocused = false;
    if (rowElement) {
      rowElement.setAttribute('aria-selected', 'false');
    }
  }

  onMount(() => {
    if (rowElement) {
      rowElement.setAttribute('tabindex', tabindex);
    }
  });

  onDestroy(() => {
    // Cleanup if needed
  });
</script>

<div
  role={role}
  tabindex={tabindex}
  aria-label={item.name || item.title || ''}
  aria-selected={isFocused ? 'true' : 'false'}
  aria-expanded={isFocused ? 'true' : 'false'}
  aria-controls={isFocused && item.id ? `details-${item.id}` : ''}
  aria-describedby={isFocused && item.description ? `desc-${item.id}` : ''}
  class="row {isFocused ? 'focused' : ''}"
  on:click={handleClick}
  on:keydown={handleKeyDown}
  on:focus={handleFocus}
  on:blur={handleBlur}
  bind:this={rowElement}
>
  <slot {item} {index} />
</div>
EOF

<script>
  import { onMount, onDestroy } from 'svelte';
  import { createEventDispatcher } from 'svelte';

  export let player;
  export let isOpen = false;
  export let onClose;
  export let role = 'dialog';
  export let 'aria-modal' = 'true';
  export let 'aria-label' = 'Player details';
  export let 'aria-describedby' = '';

  const dispatch = createEventDispatcher();

  let panelElement;
  let isClosing = false;

  function handleKeyDown(event) {
    switch (event.key) {
      case 'Escape':
        event.preventDefault();
        if (isOpen && onClose) {
          isClosing = true;
          setTimeout(() => {
            onClose();
            isClosing = false;
          }, 150); // Small delay to prevent immediate reopen
        }
        break;
      case 'Tab':
        // Handle tab navigation within dialog
        if (!event.shiftKey && document.activeElement === panelElement) {
          event.preventDefault();
          // Focus first focusable element or close
          dispatch('navigate', { direction: 'next' });
        }
        break;
    }
  }

  function handleClose() {
    if (onClose) {
      onClose();
    }
    dispatch('close');
  }

  onMount(() => {
    if (panelElement) {
      panelElement.setAttribute('role', role);
      panelElement.setAttribute('aria-modal', 'true');
      panelElement.setAttribute('aria-label', 'Player details');
      panelElement.setAttribute('tabindex', '-1');
    }
  });

  onDestroy(() => {
    // Cleanup if needed
  });
</script>

{#if isOpen && !isClosing}
  <div
    role={role}
    aria-modal={aria-modal}
    aria-label={aria-label}
    aria-describedby={aria-describedby}
    class="player-detail {isOpen ? 'open' : 'closed'}"
    on:keydown={handleKeyDown}
    bind:this={panelElement}
  >
    <div class="player-detail-backdrop" on:click={handleClose}></div>
    <div class="player-detail-content">
      <div class="player-detail-header">
        <h2>{player.name}</h2>
        <button class="close-button" aria-label="Close player details" on:click={handleClose}>
          <span aria-hidden="true">&times;</span>
        </button>
      </div>
      <div class="player-detail-body">
        <slot {player} />
      </div>
    </div>
  </div>
{/if}
EOF
