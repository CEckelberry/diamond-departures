import { writable } from 'svelte/store';

interface SoundSettings {
  enabled: boolean;
  volume: number;
}

const STORAGE_KEY = 'flap-sound-settings';

function loadFromStorage(): SoundSettings {
  if (typeof window === 'undefined') {
    return { enabled: false, volume: 0.3 };
  }
  try {
    const stored = localStorage.getItem(STORAGE_KEY);
    if (stored) {
      const parsed = JSON.parse(stored);
      // Validate basic structure
      if (typeof parsed.enabled === 'boolean' && typeof parsed.volume === 'number') {
        return parsed;
      }
    }
  } catch (e) {
    console.warn('Failed to load sound settings from localStorage', e);
  }
  return { enabled: false, volume: 0.3 };
}

function saveToStorage(settings: SoundSettings) {
  if (typeof window === 'undefined') return;
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(settings));
  } catch (e) {
    console.warn('Failed to save sound settings to localStorage', e);
  }
}

const initialSettings = loadFromStorage();

export const soundSettings = writable<SoundSettings>(initialSettings, (set) => {
  // Subscribe to changes and persist
  let current = initialSettings;
  return (value: SoundSettings) => {
    current = value;
    set(value);
    saveToStorage(value);
  };
});

export function toggleSound() {
  soundSettings.update((s) => ({ ...s, enabled: !s.enabled }));
}

export function setVolume(volume: number) {
  soundSettings.update((s) => ({ ...s, volume: Math.max(0, Math.min(1, volume)) }));
}
```

// Simple synthesized flap sound using Web Audio API to avoid external assets
// This creates a short, percussive "clack" noise.

let audioContext: AudioContext | null = null;
let lastPlayedTime = 0;
const DEBOUNCE_MS = 100;

function getAudioContext(): AudioContext {
  if (!audioContext) {
    audioContext = new (window.AudioContext || (window as any).webkitAudioContext)();
  }
  return audioContext;
}

function createFlapSound(isMany: boolean): void {
  const ctx = getAudioContext();
  const now = ctx.currentTime;

  // Duration: longer for "many" flips
  const duration = isMany ? 0.15 : 0.08;
  
  // Create buffer for noise
  const bufferSize = ctx.sampleRate * duration;
  const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
  const data = buffer.getChannelData(0);

  // Generate white noise with exponential decay
  for (let i = 0; i < bufferSize; i++) {
    const t = i / bufferSize;
    // Envelope: sharp attack, exponential decay
    const envelope = Math.exp(-t * (isMany ? 10 : 20));
    data[i] = (Math.random() * 2 - 1) * envelope;
  }

  const source = ctx.createBufferSource();
  source.buffer = buffer;

  // Filter to make it sound more like a mechanical flap
  const filter = ctx.createBiquadFilter();
  filter.type = 'bandpass';
  filter.frequency.value = isMany ? 800 : 1200;
  filter.Q.value = 1;

  // Gain node for volume control
  const gainNode = ctx.createGain();
  gainNode.gain.value = 0.5; // Base volume, adjusted by store later if needed

  source.connect(filter);
  filter.connect(gainNode);
  gainNode.connect(ctx.destination);

  source.start(now);
}

export class FlapSoundManager {
  private enabled: boolean = false;
  private volume: number = 0.3;

  playFlip(isMany: boolean): void {
    if (!this.enabled) return;

    const now = Date.now();
    if (now - lastPlayedTime < DEBOUNCE_MS) return;
    lastPlayedTime = now;

    // Adjust volume based on store setting
    // Note: In a real app, we might pass volume to the synth or use a master gain node
    // For simplicity, we'll just play the sound. The volume is handled by the store
    // if we were using HTML5 Audio elements. With Web Audio API, we'd need a master gain.
    // Let's add a master gain node for proper volume control.
    
    this._playWithVolume(isMany);
  }

  private _playWithVolume(isMany: boolean): void {
    const ctx = getAudioContext();
    const now = ctx.currentTime;

    const duration = isMany ? 0.15 : 0.08;
    const bufferSize = ctx.sampleRate * duration;
    const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
    const data = buffer.getChannelData(0);

    for (let i = 0; i < bufferSize; i++) {
      const t = i / bufferSize;
      const envelope = Math.exp(-t * (isMany ? 10 : 20));
      data[i] = (Math.random() * 2 - 1) * envelope;
    }

    const source = ctx.createBufferSource();
    source.buffer = buffer;

    const filter = ctx.createBiquadFilter();
    filter.type = 'bandpass';
    filter.frequency.value = isMany ? 800 : 1200;
    filter.Q.value = 1;

    // Master gain for volume control
    const masterGain = ctx.createGain();
    masterGain.gain.value = this.volume;

    source.connect(filter);
    filter.connect(masterGain);
    masterGain.connect(ctx.destination);

    source.start(now);
  }

  updateSettings(enabled: boolean, volume: number): void {
    this.enabled = enabled;
    this.volume = volume;
  }
}

export const flapSoundManager = new FlapSoundManager();
```

<script lang="ts">
  import { createEventDispatcher } from 'svelte';
  import type { CellProps } from './Cell.svelte';
  import { soundSettings } from '$lib/stores/sound';
  import { flapSoundManager } from '$lib/audio/flap';

  export let char: string = '';
  export let isFlipping: boolean = false;
  export let isTarget: boolean = false;
  export let isCorrect: boolean | null = null;

  const prefersReducedMotion = typeof window !== 'undefined' && 
    window.matchMedia('(prefers-reduced-motion: reduce)').matches;

  let currentChar = char;
  let isAnimating = false;

  function flip() {
    isFlipping = !isFlipping;
    if (!prefersReducedMotion && $soundSettings.enabled) {
      flapSoundManager.updateSettings($soundSettings.enabled, $soundSettings.volume);
      flapSoundManager.playFlip(isFlipping);
    }
  }

  $: if (!prefersReducedMotion) {
    flapSoundManager.updateSettings($soundSettings.enabled, $soundSettings.volume);
  }
</script>

</div>
