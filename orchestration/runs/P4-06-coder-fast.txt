+export interface SSEHandlers<T> {
+	onOpen?: () => void;
+	onMessage?: (data: T) => void;
+	onClose?: () => void;
+	onError?: (err: Event) => void;
+}
+
+export function openBoardStream<T>(
+	view: string,
+	sort: string,
+	handlers: SSEHandlers<T>
+) {
+	let source: EventSource | null = null;
+	let reconnectAttempts = 0;
+	const maxDelay = 30000;
+
+	const connect = () => {
+		const params = new URLSearchParams({ view, sort });
+		const url = `/api/board/sse?${params.toString()}`;
+
+		source = new EventSource(url);
+
+		source.onopen = () => {
+			reconnectAttempts = 0;
+			handlers.onOpen?.();
+		};
+
+		source.onmessage = (event) => {
+			try {
+				const data = JSON.parse(event.data) as T;
+				handlers.onMessage?.(data);
+			} catch (e) {
+				console.error('SSE Parse Error', e);
+			}
+		};
+
+		source.onerror = (err) => {
+			handlers.onError?.(err);
+			if (source && source.readyState === EventSource.CLOSED) {
+				source.close();
+				const delay = Math.min(Math.pow(2, reconnectAttempts) * 1000, maxDelay);
+				reconnectAttempts++;
+				setTimeout(connect, delay);
+			}
+		};
+	};
+
+	connect();
+
+	return {
+		stop: () => {
+			if (source) {
+				source.close();
+				source = null;
+			}
+		}
+	};
+}

+import { writable } from 'svelte/store';

+export interface BoardRow {
+	id: string;
+	[key: string]: any;
+}
+
+interface BoardState {
+	rows: BoardRow[];
+	loading: boolean;
+}
+
+const initialState: BoardState = {
+	rows: [],
+	loading: true
+};
+
+export const boardStore = writable<BoardState>(initialState);
+
+export function applySnapshot(rows: BoardRow[]) {
+	boardStore.set({
+		rows,
+		loading: false
+	});
+}
+
+export function applyDelta(delta: { rows: BoardRow[]; type: 'update' | 'delete' }) {
+	boardStore.update((state) => {
+		if (delta.type === 'delete') {
+			const idsToDelete = new Set(delta.rows.map((r: any) => r.id));
+			return {
+				...state,
+				rows: state.rows.filter((r) => !idsToDelete.has(r.id))
+			};
+		}
+
+		// Merge update: replace existing or append
+		const existingIds = new Set(state.rows.map((r) => r.id));
+		const newRows = delta.rows.map((r) => {
+			if (existingIds.has(r.id)) {
+				return { ...state.rows.find((row) => row.id === r.id), ...r };
+			}
+			return r;
+		});
+
+		// Simple append for new items not in existing set (optimization: usually delta contains full updated row)
+		// For robustness, we rebuild the array based on the delta rows if they are authoritative
+		// Assuming delta.rows contains the full set of changed rows.
+		// If delta is "append only", we might need different logic, but standard SSE usually sends full entity updates.
+		
+		// Re-merge: Keep state rows, update with delta rows
+		const merged = [...state.rows];
+		delta.rows.forEach((r) => {
+			const idx = merged.findIndex((row) => row.id === r.id);
+			if (idx !== -1) {
+				merged[idx] = r;
+			} else {
+				merged.push(r);
+			}
+		});
+
+		return { ...state, rows: merged };
+	});
+}
+
+export function toBoardRows(data: any[]): BoardRow[] {
+	if (!data) return [];
+	return data.map((item) => ({ id: item.id || Math.random().toString(), ...item }));
+}
