### 1. Concise Plan

1.  **Define Types**: Create `src/lib/types/board.ts` to strictly type the API response and internal `BoardRow` structure.
2.  **Implement Load Function**: Create `src/routes/+page.ts`. This function will:
    *   Extract `view` and `sort` from URL search params, validating them against allowed enums (defaulting to safe values).
    *   Fetch `/api/board` with these query parameters.
    *   Transform the raw API response into the internal `BoardRow` format required by the UI component.
    *   Return the data and a `pending` signal (handled by SvelteKit's default loading state, but we will explicitly manage the skeleton visibility in the Svelte file).
3.  **Update UI Component**: Modify `src/routes/+page.svelte`.
    *   Import the new types and the `Board` component.
    *   Accept `data` from the load function.
    *   Implement a `loading` store or use SvelteKit's `page.status`/`page.url` changes to toggle the `.board-skeleton` class.
    *   Render the skeleton if data is not yet available or if the page is transitioning.
    *   Render the `<Board>` component with the mapped rows when data is ready.

+import type { PageLoad } from './$types';
+import type { BoardResponse, BoardRow } from '$lib/types/board';
+
+const VALID_VIEWS = ['overall', 'team', 'player'];
+const VALID_SORTS = ['rank_asc', 'stat_desc', 'name_asc'];
+
+function mapToBoardRows(entries: BoardResponse['entries']): BoardRow[] {
+	return entries.map((entry) => ({
+		rank: entry.rank,
+		playerId: entry.player.id,
+		playerName: entry.player.name,
+		teamAbbr: entry.player.team_abbr,
+		position: entry.player.position,
+		statValue: entry.stat_value
+	}));
+}
+
+export const load: PageLoad = async ({ url }) => {
+	const view = VALID_VIEWS.includes(url.searchParams.get('view') || '')
+		? (url.searchParams.get('view') as BoardResponse['view'])
+		: 'overall';
+
+	const sort = VALID_SORTS.includes(url.searchParams.get('sort') || '')
+		? (url.searchParams.get('sort') as BoardResponse['sort'])
+		: 'stat_desc';
+
+	const res = await fetch(`/api/board?view=${view}&sort=${sort}`, {
+		headers: {
+			'Accept': 'application/json'
+		}
+	});
+
+	if (!res.ok) {
+		throw error(res.status, 'Failed to load board data');
+	}
+
+	const data: BoardResponse = await res.json();
+	return {
+		boardRows: mapToBoardRows(data.entries),
+		view,
+		sort
+	};
+};
