# apps/ingest/app/mlb_client.py
import asyncio
import logging
from typing import Any, Dict, Optional

import httpx

logger = logging.getLogger(__name__)


class MLBClientError(Exception):
    """Base exception for MLB Client errors."""
    pass


class MLBClient:
    """Wrapper for MLB Stats API with retry and backoff logic."""

    def __init__(
        self,
        api_key: str,
        base_url: str = "https://statsapi.mlb.com/api/v1",
        max_retries: int = 3,
        retry_base_delay: float = 1.0,
    ):
        self.api_key = api_key
        self.base_url = base_url.rstrip("/")
        self.max_retries = max_retries
        self.retry_base_delay = retry_base_delay
        self._client = httpx.AsyncClient(
            base_url=self.base_url,
            headers={
                "Authorization": f"Bearer {self.api_key}",
                "Content-Type": "application/json",
            },
            timeout=30.0,
        )

    async def get_schedule(self, date: str) -> Dict[str, Any]:
        """Fetch schedule for a specific date."""
        url = f"/schedule?date={date}"
        return await self._request_with_retry("GET", url)

    async def get_game_data(self, game_pk: int) -> Dict[str, Any]:
        """Fetch detailed game data for a specific game."""
        url = f"/game/{game_pk}/feed/live"
        return await self._request_with_retry("GET", url)

    async def _request_with_retry(
        self, method: str, url: str, **kwargs
    ) -> Dict[str, Any]:
        """Execute request with exponential backoff on 429/5xx."""
        last_exception = None
        
        for attempt in range(self.max_retries + 1):
            try:
                response = await self._client.request(method, url, **kwargs)
                
                # Handle 429 Too Many Requests
                if response.status_code == 429:
                    retry_after = response.headers.get("Retry-After")
                    delay = float(retry_after) if retry_after else self.retry_base_delay * (2 ** attempt)
                    logger.warning(f"Rate limited (429). Retrying in {delay}s (attempt {attempt + 1})")
                    await asyncio.sleep(delay)
                    continue
                
                # Handle 5xx Server Errors
                if response.status_code >= 500:
                    delay = self.retry_base_delay * (2 ** attempt)
                    logger.warning(f"Server error ({response.status_code}). Retrying in {delay}s (attempt {attempt + 1})")
                    await asyncio.sleep(delay)
                    continue
                
                # Success
                response.raise_for_status()
                return response.json()
                
            except httpx.HTTPStatusError as e:
                if e.response.status_code == 429 or e.response.status_code >= 500:
                    last_exception = e
                    delay = self.retry_base_delay * (2 ** attempt)
                    logger.warning(f"HTTP Error ({e.response.status_code}). Retrying in {delay}s (attempt {attempt + 1})")
                    await asyncio.sleep(delay)
                    continue
                raise MLBClientError(f"HTTP error: {e}") from e
            except httpx.RequestError as e:
                last_exception = e
                delay = self.retry_base_delay * (2 ** attempt)
                logger.warning(f"Request error: {e}. Retrying in {delay}s (attempt {attempt + 1})")
                await asyncio.sleep(delay)
                continue
        
        raise MLBClientError(f"Failed after {self.max_retries} retries") from last_exception

    async def close(self):
        """Close the underlying HTTP client."""
        await self._client.aclose()
```

# apps/ingest/app/job.py
import asyncio
import logging
from datetime import datetime, timedelta
from typing import Any, Dict, List

from app.config import Settings
from app.mlb_client import MLBClient, MLBClientError
from app.store import IngestStore, StoreError

logger = logging.getLogger(__name__)


async def run_ingest_job(settings: Settings) -> None:
    """
    Main entrypoint for the one-shot ingest job.
    
    1. Initialize store.
    2. Fetch data from MLB API.
    3. Insert data into store.
    4. Cleanup.
    """
    client = None
    store = None
    
    try:
        # 1. Initialize Store
        store = IngestStore(db_url=settings.db_url)
        await store.initialize()
        
        # 2. Initialize MLB Client
        client = MLBClient(
            api_key=settings.mlb_api_key,
            base_url=settings.mlb_api_base_url,
            max_retries=settings.max_retries,
            retry_base_delay=settings.retry_base_delay,
        )
        
        # 3. Fetch Data
        # Example: Ingest yesterday's schedule
        target_date = (datetime.now() - timedelta(days=1)).strftime("%Y-%m-%d")
        logger.info(f"Fetching schedule for {target_date}")
        
        schedule_data = await client.get_schedule(target_date)
        games = schedule_data.get("dates", [{}])[0].get("games", []) if schedule_data.get("dates") else []
        
        logger.info(f"Fetched {len(games)} games.")
        
        # 4. Insert Data
        if games:
            inserted_count = await store.insert_games(games)
            logger.info(f"Successfully inserted {inserted_count} games.")
        else:
            logger.info("No games found for the specified date.")
            
    except (MLBClientError, StoreError) as e:
        logger.error(f"Ingest job failed: {e}")
        raise
    finally:
        # 5. Cleanup
        if client:
            await client.close()
        if store:
            await store.close()
```

import pytest
from unittest.mock import AsyncMock, patch
from app.mlb_client import MLBClient, MLBClientError


@pytest.fixture
def client():
    return MLBClient(
        api_key="test_key",
        base_url="https://example.com/api",
        max_retries=1,
        retry_base_delay=0.01,
    )


@pytest.mark.asyncio
async def test_get_schedule_success(client):
    """Test successful schedule fetch."""
    mock_response = AsyncMock()
    mock_response.status_code = 200
    mock_response.json.return_value = {"dates": [{"games": [{"gamePk": 1}]}]}
    
    with patch.object(client._client, 'request', new_callable=AsyncMock) as mock_request:
        mock_request.return_value = mock_response
        
        result = await client.get_schedule("2023-01-01")
        
        assert result == {"dates": [{"games": [{"gamePk": 1}]}]}
        mock_request.assert_called_once_with("GET", "/schedule?date=2023-01-01")


@pytest.mark.asyncio
async def test_get_schedule_429_retry(client):
    """Test that 429 errors trigger retries."""
    mock_response_429 = AsyncMock()
    mock_response_429.status_code = 429
    mock_response_429.headers = {"Retry-After": "0.01"}
    
    mock_response_200 = AsyncMock()
    mock_response_200.status_code = 200
    mock_response_200.json.return_value = {"dates": []}
    
    with patch.object(client._client, 'request', new_callable=AsyncMock) as mock_request:
        mock_request.side_effect = [mock_response_429, mock_response_200]
        
        result = await client.get_schedule("2023-01-01")
        
        assert result == {"dates": []}
        assert mock_request.call_count == 2


@pytest.mark.asyncio
async def test_get_schedule_max_retries_exceeded(client):
    """Test that MLBClientError is raised after max retries."""
    mock_response_500 = AsyncMock()
    mock_response_500.status_code = 500
    
    with patch.object(client._client, 'request', new_callable=AsyncMock) as mock_request:
        mock_request.side_effect = mock_response_500
        
        with pytest.raises(MLBClientError):
            await client.get_schedule("2023-01-01")
        
        assert mock_request.call_count == 2  # Initial + 1 retry
