### File: `src/utils/dateUtils.ts` (New or Updated)

```typescript
export const calculateCountdown = (): string => {
  // Mocked for P6-01: Assume next season starts in 30 days from now
  const now = new Date();
  const nextSeason = new Date(now.getTime() + 30 * 24 * 60 * 60 * 1000);
  const diff = nextSeason.getTime() - now.getTime();
  
  const days = Math.floor(diff / (1000 * 60 * 60 * 24));
  return `${days} days`;
};
```
