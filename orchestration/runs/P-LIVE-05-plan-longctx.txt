# Implementation Plan for Packet P-LIVE-05

## 1. Plan with Assumptions & Risks

### Assumptions
- Existing scanner module exposes `scan()` method returning list of items
- Project uses Python 3.8+ with standard logging
- Environment variables control cadence (e.g., `SCANNER_LOOP_INTERVAL=30`)
- JSON serialization is acceptable for ops reporting
- No external dependencies beyond standard library

### Risks
- **Risk 1**: Scanner `scan()` method may block indefinitely → Add timeout and retry logic
- **Risk 2**: JSON report may grow unbounded → Add size limit and rotation
- **Risk 3**: Loop may consume excessive CPU → Add configurable sleep between iterations
- **Risk 4**: Graceful shutdown on SIGINT/SIGTERM → Add signal handlers

### Edge Cases
- Scanner returns empty list → Still log report with empty array
- Network failure → Retry with exponential backoff
- Invalid JSON output → Log error and continue
- Negative or zero interval → Default to 30 seconds

import json
import logging

import signal
import sys
import time
from typing import List, Optional

import requests
from tenacity import retry, stop_after_attempt, wait_exponential

from .scanner import scan

logger = logging.getLogger(__name__)

DEFAULT_INTERVAL = 30  # seconds
REPORT_FILE = os.getenv("SCANNER_REPORT_FILE", "/tmp/scanner_report.json")
MAX_REPORT_SIZE = int(os.getenv("SCANNER_REPORT_MAX_SIZE", 1048576))  # 1MB


class ScannerLoop:
    def __init__(self, interval: int = DEFAULT_INTERVAL):
        self.interval = interval
        self.running = False
        self.report_file = REPORT_FILE
        
    def _setup_signal_handlers(self):
        def signal_handler(signum, frame):
            logger.info(f"Received signal {signum}, shutting down gracefully")
            self.running = False
            
        signal.signal(signal.SIGINT, signal_handler)
        signal.signal(signal.SIGTERM, signal_handler)
    
    @retry(stop=stop_after_attempt(3), wait=wait_exponential(multiplier=1, min=1, max=10))
    def _scan_once(self) -> List[dict]:
        try:
            items = scan()
            logger.debug(f"Scanner returned {len(items)} items")
            return items
        except Exception as e:
            logger.error(f"Scanner failed: {e}")
            raise
    
    def _write_report(self, items: List[dict]) -> None:
        report = {
            "timestamp": time.time(),
            "items": items,
            "metadata": {
                "interval": self.interval,
                "hostname": os.uname().nodename
            }
        }
        
        # Check file size before writing
        if os.path.exists(self.report_file):
            current_size = os.path.getsize(self.report_file)
            if current_size + len(json.dumps(report)) > MAX_REPORT_SIZE:
                # Simple rotation: keep last 5 files
                import glob
                files = sorted(glob.glob(f"{self.report_file}.*"))
                for old_file in files[:-5]:
                    os.remove(old_file)
        
        with open(self.report_file, "w") as f:
            json.dump(report, f)
            f.write("\n")  # JSONL format
    
    def run(self) -> None:
        self.running = True
        self._setup_signal_handlers()
        
        logger.info(f"Starting scanner loop with {self.interval}s interval")
        
        while self.running:
            try:
                items = self._scan_once()
                self._write_report(items)
            except Exception as e:
                logger.error(f"Scanner loop iteration failed: {e}")
            
            if self.running:
                time.sleep(self.interval)
        
        logger.info("Scanner loop stopped")


def main():
    interval = int(os.getenv("SCANNER_LOOP_INTERVAL", DEFAULT_INTERVAL))
    loop = ScannerLoop(interval=interval)
    loop.run()


if __name__ == "__main__":
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s - %(name)s - %(levelname)s - %(message)s"
    )
    main()
