+from apps.ingest.constants import POSITION_TAXONOMY, QUALIFICATION_THRESHOLDS
+
+
+def resolve_primary_position(primary_raw: str, eligible_raw: str) -> str:
+    """Resolve primary position based on taxonomy."""
+    # If primary is DH or empty, fallback to first eligible
+    if not primary_raw or primary_raw == "DH":
+        return resolve_first_eligible(eligible_raw)
+    
+    mapped = POSITION_TAXONOMY.get(primary_raw)
+    if mapped:
+        return mapped
+    return primary_raw  # Fallback for unknowns
+
+
+def resolve_eligibility(eligible_raw: str) -> list[str]:
+    """Map raw eligible positions to taxonomy."""
+    if not eligible_raw:
+        return []
+    
+    positions = [p.strip() for p in eligible_raw.split("/")]
+    mapped = []
+    for p in positions:
+        tax = POSITION_TAXONOMY.get(p)
+        if tax is not None:
+            mapped.append(tax)
+    return mapped
+
+
+def resolve_first_eligible(eligible_raw: str) -> str:
+    """Get the first valid eligible position."""
+    if not eligible_raw:
+        return "UT"  # Default fallback
+    
+    positions = [p.strip() for p in eligible_raw.split("/")]
+    for p in positions:
+        tax = POSITION_TAXONOMY.get(p)
+        if tax is not None:
+            return tax
+    return "UT"
+
+
+def determine_pitcher_role(starts: int, games: int) -> str:
+    """Determine SP or RP based on start ratio."""
+    if games == 0:
+        return "RP"
+    
+    ratio = starts / games
+    if ratio > QUALIFICATION_THRESHOLDS["SP_START_RATIO"]:
+        return "SP"
+    return "RP"
+
+
+def check_qualification(player_type: str, stats: dict) -> bool:
+    """Check if player meets qualification thresholds."""
+    if player_type == "HITTER":
+        return stats.get("games", 0) >= QUALIFICATION_THRESHOLDS["HITTER_GAMES"]
+    elif player_type == "SP":
+        return stats.get("innings_pitched", 0) >= QUALIFICATION_THRESHOLDS["SP_IPS"]
+    elif player_type == "RP":
+        return stats.get("innings_pitched", 0) >= QUALIFICATION_THRESHOLDS["RP_IPS"]
+    return False
