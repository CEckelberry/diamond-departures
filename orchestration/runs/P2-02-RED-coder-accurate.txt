import pytest
from fastapi.testclient import TestClient
from unittest.mock import patch, MagicMock
from datetime import datetime, timezone

# Import the app instance (assuming standard FastAPI structure)
from apps.api.main import app

client = TestClient(app)


@pytest.fixture
def mock_leaderboard_service():
    """Mock the leaderboard service to avoid DB access."""
    with patch("apps.api.services.leaderboard_service.get_board_data") as mock_service:
        # Define a deterministic response for valid queries
        mock_response = {
            "players": [
                {
                    "id": 1,
                    "name": "Player One",
                    "team": "Team A",
                    "wrc_plus": 150.0,
                    "metadata": {"position": "OF", "age": 25}
                },
                {
                    "id": 2,
                    "name": "Player Two",
                    "team": "Team B",
                    "wrc_plus": 140.0,
                    "metadata": {"position": "SS", "age": 30}
                }
            ],
            "freshness": {
                "timestamp": datetime.now(timezone.utc).isoformat(),
                "age_category": "live"
            }
        }
        mock_service.return_value = mock_response
        yield mock_service


class TestBoardEndpoint:
    """Tests for GET /api/board endpoint."""

    def test_get_board_valid_view_sort(self, mock_leaderboard_service):
        """
        Goal 1: GET /api/board?view=hitters&sort=wRC+ returns <=100 ranked entries with player metadata.
        """
        response = client.get("/api/board", params={"view": "hitters", "sort": "wRC+"})
        
        assert response.status_code == 200
        data = response.json()
        
        # Check structure
        assert "players" in data
        assert "freshness" in data
        
        # Check count constraint
        assert len(data["players"]) <= 100
        assert len(data["players"]) > 0  # Ensure mock returned data
        
        # Check metadata presence
        for player in data["players"]:
            assert "id" in player
            assert "name" in player
            assert "metadata" in player

    def test_get_board_invalid_view(self, mock_leaderboard_service):
        """
        Goal 2: Invalid view returns 400 with clear error.
        """
        response = client.get("/api/board", params={"view": "invalid_view"})
        
        assert response.status_code == 400
        data = response.json()
        assert "detail" in data
        # Check for a clear error message regarding the view parameter
        assert "view" in data["detail"].lower() or "invalid" in data["detail"].lower()

    def test_get_board_invalid_sort(self, mock_leaderboard_service):
        """
        Goal 3: Invalid sort returns 400 with clear error.
        """
        response = client.get("/api/board", params={"view": "hitters", "sort": "invalid_sort"})
        
        assert response.status_code == 400
        data = response.json()
        assert "detail" in data
        # Check for a clear error message regarding the sort parameter
        assert "sort" in data["detail"].lower() or "invalid" in data["detail"].lower()

    def test_get_board_freshness_object(self, mock_leaderboard_service):
        """
        Goal 4: Freshness object includes timestamp and age category.
        """
        response = client.get("/api/board", params={"view": "hitters", "sort": "wRC+"})
        
        assert response.status_code == 200
        data = response.json()
        
        freshness = data.get("freshness")
        assert freshness is not None
        
        assert "timestamp" in freshness
        assert "age_category" in freshness
        
        # Validate age_category values
        valid_categories = ["live", "recent", "stale", "old"]
        assert freshness["age_category"] in valid_categories
        
        # Validate timestamp format (ISO 8601)
        try:
            datetime.fromisoformat(freshness["timestamp"])
        except ValueError:
            pytest.fail("Timestamp is not in valid ISO 8601 format")
