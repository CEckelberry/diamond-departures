import assert from 'node:assert/strict';

const ROOT_DIR = path.resolve(__dirname, '../..'); // Assumes test is in apps/web/tests, root is apps/web
// Adjust based on actual repo structure. If apps/web is the project root for these tests:
const PROJECT_ROOT = path.resolve(__dirname, '..'); 

describe('Player Detail URL Routing', () => {
  const routeDir = path.join(PROJECT_ROOT, 'src/routes/player/[slug]');
  const pageSveltePath = path.join(routeDir, '+page.svelte');
  const pageTsPath = path.join(routeDir, '+page.ts');

  it('should have route files: +page.svelte and +page.ts', () => {
    assert.ok(fs.existsSync(pageSveltePath), `Missing ${pageSveltePath}`);
    assert.ok(fs.existsSync(pageTsPath), `Missing ${pageTsPath}`);
  });

  it('should import and use Board and Panel components in +page.svelte', () => {
    const content = fs.readFileSync(pageSveltePath, 'utf-8');
    
    // Check for imports or usage of Board and Panel
    // Assuming standard Svelte component names
    assert.ok(
      /<Board\s/.test(content) || /import\s+Board/.test(content),
      '+page.svelte must use or import Board component'
    );
    assert.ok(
      /<Panel\s/.test(content) || /import\s+Panel/.test(content),
      '+page.svelte must use or import Panel component'
    );
  });

  it('should derive selected player from slug and entries in +page.ts', () => {
    const content = fs.readFileSync(pageTsPath, 'utf-8');
    
    // Check for usage of 'slug' and 'entries' variables
    assert.ok(
      /slug/.test(content),
      '+page.ts must reference slug'
    );
    assert.ok(
      /entries/.test(content),
      '+page.ts must reference entries'
    );
    // Check for player assignment or return
    assert.ok(
      /player/.test(content),
      '+page.ts must define or return player'
    );
  });

  it('should have a back/close navigation path to / in +page.svelte', () => {
    const content = fs.readFileSync(pageSveltePath, 'utf-8');
    
    // Check for link to root
    assert.ok(
      /href=["']\/["']/.test(content) || /to=["']\/["']/.test(content),
      '+page.svelte must have a navigation element pointing to /'
    );
  });
});
