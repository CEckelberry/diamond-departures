```python
# apps/api/boards/memory_store.py
import time
from typing import List, Dict, Any

from .store import BoardStore


class InMemoryBoardStore(BoardStore):
    def __init__(self, data: List[Dict[str, Any]] | None = None, freshness_age: int = 0):
        self._data = data or []
        self._freshness_age = freshness_age

    async def get_top_100(self) -> List[Dict[str, Any]]:
        return self._data[:100]

    async def get_freshness_age(self) -> int:
        return self._freshness_age
```

#### New File: `apps/api/boards/handler.py`
```python
# apps/api/boards/handler.py
from fastapi import APIRouter, Depends, HTTPException
from typing import List, Dict, Any

from .store import BoardStore

router = APIRouter()


def get_board_store() -> BoardStore:
    """Dependency to inject the board store."""
    # In a real app, this would be injected via DI container.
    # For this packet, we assume the app factory wires this up or
    # we rely on a global singleton for simplicity in tests.
    # However, to make it testable and clean, we'll define a default
    # that can be overridden.
    from .memory_store import InMemoryBoardStore
    return InMemoryBoardStore()


@router.get("/api/board")
async def get_leaderboard(store: BoardStore = Depends(get_board_store)):
    """
    GET /api/board
    Returns top 100 leaderboard and freshness info.
    """
    top_100 = await store.get_top_100()
    freshness_age = await store.get_freshness_age()

    # Categorize freshness
    if freshness_age < 60:
        freshness_category = "fresh"
    elif freshness_age < 3600:
        freshness_category = "stale"
    else:
        freshness_category = "very_stale"

    return {
        "leaderboard": top_100,
        "freshness": {
            "age_seconds": freshness_age,
            "category": freshness_category
        }
    }
```
