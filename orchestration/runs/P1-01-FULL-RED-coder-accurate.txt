"""Tests for wRC+ calculation and LeagueContext."""

import pytest

try:
    from ..wrc import wrc_plus, LeagueContext
except ImportError:
    pytest.skip("Implementation not yet available", allow_module_level=True)


class TestLeagueContext:
    def test_required_fields(self):

        lc = LeagueContext(
            lgwOBA=0.320,
            wOBAScale=0.9,
            lgRunsPerPA=0.10,
            parkFactor=1.0
        )
        assert lc.lgwOBA == 0.320
        assert lc.wOBAScale == 0.9
        assert lc.lgRunsPerPA == 0.10
        assert lc.parkFactor == 1.0

    def test_missing_field_raises_error(self):
        """Test that missing required field raises TypeError."""
        with pytest.raises(TypeError):
            LeagueContext(lgwOBA=0.320)


class TestWrcPlusCalculation:
    def test_hand_verified_wrc_plus(self):
        """Test wRC+ with a hand-verified sample."""
        # Player stats: 1 PA, 1 HR (wOBA = 2.10 / 1 = 2.10)
        # League Context: lgwOBA=0.320, wOBAScale=0.9, lgRunsPerPA=0.10, PF=1.0
        # Formula: 100 * ((wOBA - lgwOBA) / wOBAScale + lgRuns/PA) / (lgRuns/PA)
        # Step 1: (2.10 - 0.320) / 0.9 = 1.78 / 0.9 = 1.9777...
        # Step 2: 1.9777... + 0.10 = 2.0777...
        # Step 3: 2.0777... / 0.10 = 20.777...
        # Step 4: 100 * 20.777... = 2077.77...
        # Note: This is an extreme case. Let's use a more realistic one.
        
        # Realistic sample:
        # AB=500, BB=50, HBP=10, 1B=100, 2B=20, 3B=5, HR=20, IBB=5, SF=5
        # PA = AB + BB - IBB + SF + HBP = 500 + 50 - 5 + 5 + 10 = 560
        # wOBA Numerator:
        # BB: 50 * 0.69 = 34.5
        # HBP: 10 * 0.72 = 7.2
        # 1B: 100 * 0.89 = 89.0
        # 2B: 20 * 1.27 = 25.4
        # 3B: 5 * 1.62 = 8.1
        # HR: 20 * 2.10 = 42.0
        # Total Num = 34.5 + 7.2 + 89.0 + 25.4 + 8.1 + 42.0 = 206.2
        # wOBA = 206.2 / 560 = 0.368214...
        
        # League Context: lgwOBA=0.320, wOBAScale=0.9, lgRunsPerPA=0.10, PF=1.0
        # Step 1: (0.368214 - 0.320) / 0.9 = 0.048214 / 0.9 = 0.053571...
        # Step 2: 0.053571 + 0.10 = 0.153571...
        # Step 3: 0.153571 / 0.10 = 1.53571...
        # Step 4: 100 * 1.53571 = 153.571
        
        stats = {
            "AB": 500,
            "BB": 50,
            "HBP": 10,
            "1B": 100,
            "2B": 20,
            "3B": 5,
            "HR": 20,
            "IBB": 5,
            "SF": 5,
        }
        lc = LeagueContext(
            lgwOBA=0.320,
            wOBAScale=0.9,
            lgRunsPerPA=0.10,
            parkFactor=1.0
        )
        result = wrc_plus(stats, lc)
        assert abs(result - 153.57) < 0.1

    def test_zero_guard_in_wrc_plus(self):
        """Test that zero denominator in wRC+ returns 0 or handles gracefully."""
        stats = {
            "AB": 0,
            "BB": 0,
            "HBP": 0,
            "1B": 0,
            "2B": 0,
            "3B": 0,
            "HR": 0,
            "IBB": 0,
            "SF": 0,
        }
        lc = LeagueContext(
            lgwOBA=0.320,
            wOBAScale=0.9,
            lgRunsPerPA=0.10,
            parkFactor=1.0
        )
        result = wrc_plus(stats, lc)
        assert result == 0.0
