    assert.ok(results.includes('c'), 'Final state c should be in results');
    // If the implementation skips intermediates:

    // If it processes all:

    // For now, we assert that the queue doesn't crash and processes something.
    assert.ok(results.length > 0, 'Queue should process at least one item');
  });

  it('should preserve order of updates', async () => {
    const queue = new QueueManager();
    const results = [];

  it('should skip animation and emit flash signal when reduced motion is enabled', () => {

    // Trigger the flip
    handler.flip('new-state');
    
    // Assert that no animation was queued
    assert.strictEqual(handler.isAnimating(), false, 'Should not be animating');
    
    // Assert that flash signal was emitted
    assert.ok(flashSignal.emitted, 'Flash signal should be emitted');

  it('should animate normally when reduced motion is disabled', () => {
    const handler = new ReducedMotionHandler(false); // Disable reduced motion
