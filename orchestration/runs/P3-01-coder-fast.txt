<script lang="ts">
	export let value: string = '';
	export let width: number = 28;
	export let height: number = 36;

	// Ensure value is a single character for display
	$: displayChar = value ? value.charAt(0).toUpperCase() : ' ';
</script>

<div
	class="cell"
	style="--cell-width: {width}px; --cell-height: {height}px;"
>
	<!-- Top Half -->
	<div class="half top">
		<span>{displayChar}</span>
	</div>
	
	<!-- Hairline Separator -->
	<div class="hairline"></div>
	
	<!-- Bottom Half -->
	<div class="half bottom">
		<span>{displayChar}</span>
	</div>
</div>
