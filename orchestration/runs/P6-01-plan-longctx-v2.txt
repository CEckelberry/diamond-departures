+import { useState, useEffect } from 'react';
+
+export const useSeason = () => {
+  const [season, setSeason] = useState<'final' | 'off-season'>('final');
+
+  useEffect(() => {
+    const checkSeason = () => {
+      const currentMonth = new Date().getMonth();
+      const isOffSeason = currentMonth >= 11 || currentMonth <= 2; // Nov-Feb
+      setSeason(isOffSeason ? 'off-season' : 'final');
+    };
+
+    checkSeason();
+    const interval = setInterval(checkSeason, 60000); // Check monthly
+    return () => clearInterval(interval);
+  }, []);
+
+  return { season };
+};
