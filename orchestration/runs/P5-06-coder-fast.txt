#### File: `apps/web/src/lib/components/board/FreshnessPanel.svelte`

```svelte
<script lang="ts">
	import { onMount, onDestroy } from 'svelte';
	import { createEventDispatcher } from 'svelte';

	export let isOpen: boolean = false;

	const dispatch = createEventDispatcher();

	interface IngestRun {
		id: string;
		timestamp: string;
		status: string;
	}

	interface StatFreshness {
		stat: string;
		lastUpdated: string;
		freshnessSeconds: number;
	}

	interface FreshnessData {
		lastIngestRuns: IngestRun[];
		statFreshness: StatFreshness[];
		schemaDriftLog?: string;
	}

	let data: FreshnessData | null = null;
	let loading = false;
	let error: string | null = null;
	let panelRef: HTMLDivElement;

	onMount(() => {
		document.addEventListener('keydown', handleKeyDown);
	});

	onDestroy(() => {
		document.removeEventListener('keydown', handleKeyDown);
	});

	async function fetchData() {
		if (!isOpen) return;
		
		loading = true;
		error = null;
		data = null;

		try {
			const response = await fetch('/api/freshness');
			if (!response.ok) {
				throw new Error(`Failed to fetch freshness data: ${response.statusText}`);
			}
			data = await response.json();
		} catch (err) {
			error = err instanceof Error ? err.message : 'Unknown error occurred';
		} finally {
			loading = false;
		}
	}

	function handleKeyDown(event: KeyboardEvent) {
		if (event.key === 'Escape' && isOpen) {
			dispatch('close');
		}
	}

	// Re-fetch when opened
	$: if (isOpen) {
		fetchData();
	}

	function handleClose() {
		dispatch('close');
	}

	function formatTime(timestamp: string): string {
		try {
			const date = new Date(timestamp);
			return date.toLocaleString();
		} catch {
			return timestamp;
		}
	}

	function getFreshnessColor(seconds: number): string {
		if (seconds < 300) return 'bg-green-500'; // < 5 mins
		if (seconds < 3600) return 'bg-yellow-500'; // < 1 hour
		return 'bg-red-500'; // > 1 hour
	}
</script>

{#if isOpen}
	<div 
		class="fixed inset-0 z-50 flex items-center justify-center bg-black bg-opacity-50"
		on:click={handleClose}
		role="dialog"
		aria-modal="true"
		aria-labelledby="freshness-panel-title"
	>
		<div 
			class="bg-white rounded-lg shadow-xl w-full max-w-2xl max-h-[80vh] overflow-hidden flex flex-col"
			on:click|stopPropagation
			{panelRef}
		>
			<!-- Header -->
			<div class="flex items-center justify-between p-4 border-b">
				<h2 id="freshness-panel-title" class="text-xl font-bold text-gray-800">Freshness Debug Panel</h2>
				<button
					class="text-gray-500 hover:text-gray-700 focus:outline-none focus:ring-2 focus:ring-blue-500 rounded p-1"
					on:click={handleClose}
					aria-label="Close panel"
				>
					<svg xmlns="http://www.w3.org/2000/svg" class="h-6 w-6" fill="none" viewBox="0 0 24 24" stroke="currentColor">
						<path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M6 18L18 6M6 6l12 12" />
					</svg>
				</button>
			</div>

			<!-- Content -->
			<div class="p-4 overflow-y-auto flex-1">
				{#if loading}
					<div class="flex justify-center items-center h-32">
						<div class="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-500"></div>
					</div>
				{:else if error}
					<div class="text-red-500 text-center p-4">
						<p class="font-bold">Error</p>
						<p>{error}</p>
					</div>
				{:else if data}
					<!-- Last Ingest Runs -->
					<section class="mb-6">
						<h3 class="text-lg font-semibold text-gray-700 mb-2">Last Ingest Runs</h3>
						{#if data.lastIngestRuns.length === 0}
							<p class="text-gray-500 italic">No ingest runs found.</p>
						{:else}
							<ul class="space-y-2">
								{#each data.lastIngestRuns.slice(0, 10) as run (run.id)}
									<li class="flex justify-between items-center p-2 bg-gray-50 rounded">
										<span class="text-sm font-mono text-gray-600">{run.id}</span>
										<span class="text-sm text-gray-800">{formatTime(run.timestamp)}</span>
										<span class="text-xs px-2 py-1 rounded-full {run.status === 'success' ? 'bg-green-100 text-green-800' : 'bg-red-100 text-red-800'}">
											{run.status}
										</span>
									</li>
								{/each}
							</ul>
						{/if}
					</section>

					<!-- Per-Stat Freshness -->
					<section class="mb-6">
						<h3 class="text-lg font-semibold text-gray-700 mb-2">Per-Stat Freshness</h3>
						{#if data.statFreshness.length === 0}
							<p class="text-gray-500 italic">No stat freshness data available.</p>
						{:else}
							<div class="space-y-2">
								{#each data.statFreshness as stat (stat.stat)}
									<div class="flex items-center justify-between p-2 bg-gray-50 rounded">
										<span class="text-sm font-medium text-gray-800">{stat.stat}</span>
										<div class="flex items-center space-x-2">
											<div class="w-3 h-3 rounded-full {getFreshnessColor(stat.freshnessSeconds)}" title="{stat.freshnessSeconds} seconds ago"></div>
											<span class="text-xs text-gray-600">{Math.round(stat.freshnessSeconds / 60)}m ago</span>
										</div>
									</div>
								{/each}
							</div>
						{/if}
					</section>

					<!-- Schema Drift Log -->
					{#if data.schemaDriftLog}
						<section>
							<h3 class="text-lg font-semibold text-gray-700 mb-2">Schema Drift Log</h3>
							<div class="bg-gray-800 text-green-400 p-4 rounded font-mono text-xs overflow-x-auto">
								<pre>{data.schemaDriftLog}</pre>
							</div>
						</section>
					{/if}
				{/if}
			</div>

			<!-- Footer -->
			<div class="p-4 border-t bg-gray-50 flex justify-end">
				<button
					class="px-4 py-2 bg-blue-500 text-white rounded hover:bg-blue-600 focus:outline-none focus:ring-2 focus:ring-blue-500 focus:ring-offset-2"
					on:click={handleClose}
				>
					Close
				</button>
			</div>
		</div>
	</div>
{/if}

<style>
	/* Add any custom styles if needed, otherwise relying on Tailwind */
</style>
```
