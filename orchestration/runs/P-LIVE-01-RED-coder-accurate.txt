## 1. Plan with Assumptions + Risks

### Assumptions
1. **Language/Stack**: Python 3.9+ with `pytest`, `httpx` (for async HTTP), and `pydantic` (for config). This is the standard for modern data engineering pipelines in this context.
2. **Directory Structure**:
   - Source code lives in `src/mlb_live/`.
   - Tests live in `tests/test_live/`.
3. **API Semantics**:
   - MLB Stats API v1 is the source of truth.
   - `/api/v1/game/feed/live/{gamePk}` provides real-time state.
   - `/api/v1/game/{gamePk}/feed/live` is redundant but sometimes used; we standardize on `/api/v1/game/feed/live/{gamePk}`.
   - `/api/v1/game/changes?since={timestamp}` returns a list of `gamePk`s that have changed state since the given epoch timestamp.
4. **Provider Pattern**: A `LiveDataProvider` protocol/interface that abstracts fetching game state and changes. This allows swapping between MLB Stats API, a mock provider for testing, or a future provider (e.g., Statcast).
5. **Scanner Mode**: A background job that periodically calls `/api/v1/game/changes` to get a delta of changed games, then fetches only those games via the provider. It does *not* fetch all games every time.

### Risks & Mitigations
- **Risk**: Tests may pass if implementation is incomplete due to loose mocking.
  - **Mitigation**: Use strict `httpx` mocking with `respx` or `responses` to ensure API calls are made as expected. Fail if unexpected endpoints are hit.
- **Risk**: Time-based logic in `changes` endpoint can be flaky.
  - **Mitigation**: Mock the `time.time()` function or inject a fixed timestamp into the scanner context.
- **Risk**: Provider interface may be too rigid for future extensions.
  - **Mitigation**: Use Python's `Protocol` from `typing` for structural subtyping, allowing duck-typing flexibility.

### Edge Cases & Failure Modes
1. **Empty Changes**: `/api/v1/game/changes` returns `[]`. Scanner should do nothing and not crash.
2. **Network Timeout**: Provider fetch fails. Scanner should log error and retry or skip, not halt the entire job.
3. **Invalid Game PK**: API returns 404 for a game PK in changes. Scanner should skip that game and continue with others.
4. **Config Error**: Missing `MLB_PROVIDER_TYPE` env var. Should raise `ValueError` or use a default (e.g., `mlb_stats`).

```python
"""Scanner job that uses game changes to fetch only updated games."""

import logging
import time
from typing import List

from .providers import LiveDataProvider, create_provider
from .config import get_provider_type, get_api_base_url

logger = logging.getLogger(__name__)

class GameChangeScanner:
    """Scans for game changes and processes them."""

    def __init__(self, provider: LiveDataProvider = None):
        self.provider = provider or self._create_default_provider()

    def _create_default_provider(self) -> LiveDataProvider:
        provider_type = get_provider_type()
        api_base_url = get_api_base_url()
        kwargs = {}
        if api_base_url:
            kwargs["api_base_url"] = api_base_url
        return create_provider(provider_type, **kwargs)

    def scan(self, since_timestamp: int = None) -> List[int]:
        """Perform a scan for game changes.

        Args:
            since_timestamp: Optional timestamp to start from. If None, uses current time - 60s.

        Returns:
            A list of game PKs that were processed.
        """
        if since_timestamp is None:
            since_timestamp = int(time.time()) - 60

        try:
            changed_game_pks = self.provider.get_changes_since(since_timestamp)
        except Exception as e:
            logger.error(f"Failed to fetch game changes: {e}")
            return []

        processed_games = []
        for game_pk in changed_game_pks:
            try:
                state = self.provider.get_game_state(game_pk)
                if state is not None:
                    logger.info(f"Processed game {game_pk}")
                    processed_games.append(game_pk)
                else:
                    logger.warning(f"No state found for game {game_pk}")
            except Exception as e:
                logger.error(f"Failed to fetch state for game {game_pk}: {e}")
                # Continue with other games
                continue

        return processed_games

    def close(self):
        self.provider.close()
```

```python
"""Tests for the GameChangeScanner job."""

import pytest
from unittest.mock import MagicMock, patch
from src.mlb_live.scanner import GameChangeScanner
from src.mlb_live.providers import LiveDataProvider


class MockProvider(LiveDataProvider):
    """A mock provider for testing scanner logic."""

    def __init__(self, changes_response=None, game_states=None):
        self.changes_response = changes_response or []
        self.game_states = game_states or {}
        self.get_changes_calls = []
        self.get_game_state_calls = []

    def get_changes_since(self, since_timestamp: int):
        self.get_changes_calls.append(since_timestamp)
        return self.changes_response

    def get_game_state(self, game_pk: int):
        self.get_game_state_calls.append(game_pk)
        return self.game_states.get(game_pk)

    def close(self):
        pass


class TestGameChangeScanner:
    """Test the GameChangeScanner behavior."""

    def test_scan_with_no_changes(self):
        """Scanner should return empty list if no changes."""
        provider = MockProvider(changes_response=[])
        scanner = GameChangeScanner(provider=provider)
        result = scanner.scan(since_timestamp=1000)
        assert result == []
        assert len(provider.get_changes_calls) == 1
        assert len(provider.get_game_state_calls) == 0

    def test_scan_with_changed_games(self):
        """Scanner should fetch state for each changed game."""
        provider = MockProvider(
            changes_response=[12345, 67890],
            game_states={12345: {"gamePk": 12345}, 67890: {"gamePk": 67890}}
        )
        scanner = GameChangeScanner(provider=provider)
        result = scanner.scan(since_timestamp=1000)
        assert sorted(result) == [12345, 67890]
        assert len(provider.get_changes_calls) == 1
        assert sorted(provider.get_game_state_calls) == [12345, 67890]

    def test_scan_skips_games_with_no_state(self):
        """Scanner should skip games that return None state."""
        provider = MockProvider(
            changes_response=[12345],
            game_states={}  # No state for 12345
        )
        scanner = GameChangeScanner(provider=provider)
        result = scanner.scan(since_timestamp=1000)
        assert result == []
        assert len(provider.get_game_state_calls) == 1

    def test_scan_continues_on_error(self):
        """Scanner should continue processing other games if one fails."""
        provider = MockProvider(
            changes_response=[12345, 67890],
            game_states={67890: {"gamePk": 67890}}
        )
        # Simulate an error for game 12345
        def get_game_state_side_effect(game_pk):
            if game_pk == 12345:
                raise Exception("API Timeout")
            return provider.game_states.get(game_pk)

        provider.get_game_state = get_game_state_side_effect

        scanner = GameChangeScanner(provider=provider)
        result = scanner.scan(since_timestamp=1000)
        assert result == [67890]
        assert len(provider.get_game_state_calls) == 2

    def test_scan_uses_default_timestamp_if_none(self):
        """Scanner should use current time - 60s if since_timestamp is None."""
        provider = MockProvider(changes_response=[])
        scanner = GameChangeScanner(provider=provider)
        with patch("src.mlb_live.scanner.time.time") as mock_time:
            mock_time.return_value = 1000.0
            scanner.scan()
            # Should have called get_changes_since with 940 (1000 - 60)
            assert provider.get_changes_calls == [940]

    def test_close_calls_provider_close(self):
        """Scanner close should call provider close."""
        provider = MockProvider()
        scanner = GameChangeScanner(provider=provider)
        scanner.close()
        # No assertion needed if no error, but we can check if close was called
        # Since MockProvider.close is a pass, we just ensure no exception.
```
