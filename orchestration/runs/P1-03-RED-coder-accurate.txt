+"""Defensive statistics parsers."""
+
+from .constants import MIN_INNINGS_THRESHOLD
+
+
+def extract_defensive_metrics(payload: dict) -> dict:
+    """
+    Extract DRS, UZR, UZR150, and OAA from a raw source payload.
+    
+    Args:
+        payload: Dictionary containing raw defensive stats.
+        
+    Returns:
+        Dictionary with keys: 'drs', 'uzr', 'uzr150', 'oaa'.
+    """
+    return {
+        'drs': payload.get('drs'),
+        'uzr': payload.get('uzr'),
+        'uzr150': payload.get('uzr150'),
+        'oaa': payload.get('oaa'),
+    }
+
+
+def normalize_uzr150_from_per_game(payload: dict) -> float:
+    """
+    Normalize UZR/150 if the source provides a per-game value.
+    
+    Assumes payload contains 'uzr_per_game' and 'games_played'.
+    Standardizes to a full season (approx 162 games for MLB, but we use 
+    the actual games played in the payload context or a default factor).
+    
+    Note: UZR/150 is already normalized per 150 games. If source gives 
+    per-game, we multiply by 150 to get the standard metric.
+    """
+    uzr_per_game = payload.get('uzr_per_game')
+    if uzr_per_game is None:
+        return None
+    
+    # UZR/150 implies normalization to 150 games. 
+    # If the source gives per-game, multiply by 150.
+    return uzr_per_game * 150
+
+
+def qualify_sample(innings_played: float) -> str:
+    """
+    Tag a defensive sample based on innings played.
+    
+    Args:
+        innings_played: Number of innings played.
+        
+    Returns:
+        '(noisy)' if below threshold, else 'qualified'.
+    """
+    if innings_played < MIN_INNINGS_THRESHOLD:
+        return "(noisy)"
+    return "qualified"

+class TestExtractDefensiveMetrics:
+    def test_extracts_all_metrics(self):
+        payload = {
+            'drs': 5,
+            'uzr': 3.2,
+            'uzr150': 4.1,
+            'oaa': 8,
+        }
+        result = extract_defensive_metrics(payload)
+        assert result['drs'] == 5
+        assert result['uzr'] == 3.2
+        assert result['uzr150'] == 4.1
+        assert result['oaa'] == 8

+    def test_handles_missing_metrics(self):
+        payload = {'drs': 1}
+        result = extract_defensive_metrics(payload)
+        assert result['drs'] == 1
+        assert result['uzr'] is None
+        assert result['uzr150'] is None
+        assert result['oaa'] is None

+
+class TestNormalizeUZR150:
+    def test_normalizes_per_game_to_uzr150(self):
+        payload = {'uzr_per_game': 0.1}
+        result = normalize_uzr150_from_per_game(payload)
+        # 0.1 per game * 150 games = 15.0
+        assert result == 15.0

+    def test_handles_missing_per_game(self):
