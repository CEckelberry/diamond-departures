```python
from unittest.mock import MagicMock, patch, call

import pytest


class TestPubSubSubscriber:
    """Tests for the Pub/Sub subscriber invalidation flow."""

    @pytest.fixture
    def mock_invalidator(self):
        """Mock cache invalidation service."""
        with patch('apps.api.pubsub.CacheInvalidator') as mock:
            yield mock

    @pytest.fixture
    def mock_message(self):
        """Mock a Pub/Sub message."""
        message = MagicMock()
        message.data = b'{"views": [{"view": "user_profile", "sort": "created"}, {"view": "post_feed", "sort": "relevance"}]}'
        return message

    def test_subscriber_invalidates_keys(self, mock_invalidator, mock_message):
        """Subscriber parses refresh message and invalidates each targeted (view, sort) key."""
        from apps.api.pubsub import Subscriber
        
        subscriber = Subscriber(invalidator=mock_invalidator)
        
        # Act
        subscriber.handle_message(mock_message)
        
        # Assert
        expected_calls = [
            call.invalidate(view="user_profile", sort="created"),
            call.invalidate(view="post_feed", sort="relevance")
        ]
        mock_invalidator.invalidate.assert_has_calls(expected_calls, any_order=False)

    def test_subscriber_acks_on_success(self, mock_invalidator, mock_message):
        """Subscriber ACKs message after successful invalidation."""
        from apps.api.pubsub import Subscriber
        
        subscriber = Subscriber(invalidator=mock_invalidator)
        
        # Act
        subscriber.handle_message(mock_message)
        
        # Assert
        mock_message.ack.assert_called_once()

    def test_subscriber_nacks_on_failure(self, mock_invalidator, mock_message):
        """Subscriber NACKs message when invalidation raises."""
        from apps.api.pubsub import Subscriber
        
        # Setup: Make invalidation raise an error
        mock_invalidator.invalidate.side_effect = Exception("Cache connection failed")
        
        subscriber = Subscriber(invalidator=mock_invalidator)
        
        # Act & Assert
        with pytest.raises(Exception, match="Cache connection failed"):
            subscriber.handle_message(mock_message)
        
        # Note: Depending on implementation, nack might be called before re-raising or not at all.
        # For RED, we assume the contract requires NACK on failure.
        mock_message.nack.assert_called_once()

    def test_subscriber_logs_refreshed_keys(self, mock_invalidator, mock_message, caplog):
        """Subscriber logs refreshed keys with reason for observability."""
        import logging
        from apps.api.pubsub import Subscriber
        
        subscriber = Subscriber(invalidator=mock_invalidator)
        
        # Act
        subscriber.handle_message(mock_message)
        
        # Assert
        assert "Invalidated keys" in caplog.text or "refreshed keys" in caplog.text.lower()
        assert "user_profile" in caplog.text
        assert "post_feed" in caplog.text
        assert "created" in caplog.text
        assert "relevance" in caplog.text
```
