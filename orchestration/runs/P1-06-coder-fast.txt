    *   Return a list of normalized stat-change dictionaries.

+"""
+State diffing and idempotency helpers.
+"""
+
+import logging
+from typing import Dict, Any, List
+
+logger = logging.getLogger(__name__)
+
+
+def compute_diff(old_state: Dict[str, Any], new_state: Dict[str, Any]) -> List[str]:
+    """
+    Compute the difference between two state snapshots.
+
+    Args:
+        old_state: The previous state dictionary.
+        new_state: The current state dictionary.
+
+    Returns:
+        A list of keys that have changed values between old_state and new_state.
+    """
+    changed_keys = []
+    all_keys = set(old_state.keys()) | set(new_state.keys())
+
+    for key in all_keys:
+        old_val = old_state.get(key)
+        new_val = new_state.get(key)
+        if old_val != new_val:
+            changed_keys.append(key)
+            logger.debug(f"Key '{key}' changed from {old_val} to {new_val}")
+
+    return changed_keys
+
+
+def is_idempotent(last_event: Dict[str, Any], current_event: Dict[str, Any]) -> bool:
+    """
+    Check if the current event is a duplicate of the last processed event.
+
+    Args:
+        last_event: The previously processed event dictionary.
+        current_event: The current event dictionary to check.
+
+    Returns:
+        True if the events are identical (idempotent), False otherwise.
+    """
+    if not last_event or not current_event:
+        return False
+
+    is_same = last_event == current_event
+    logger.debug(f"Idempotency check: {is_same}")
+    return is_same
```

+"""
+Game feed processing module.
+"""
+
+import logging
+from typing import List, Dict, Any
+
+from .state_diff import compute_diff, is_idempotent
+
+logger = logging.getLogger(__name__)
+
+# Mock feed data for demonstration purposes.
+_MOCK_FEEDS = {
+    "g1": [
+        {"event_id": "e1", "player_id": "p1", "type": "stat_change", "stats": {"points": 10}},
+        {"event_id": "e2", "player_id": "p2", "type": "stat_change", "stats": {"points": 5}},
+    ],
+    "g2": [
+        {"event_id": "e3", "player_id": "p3", "type": "stat_change", "stats": {"points": 15}},
+    ],
+}
+
+
+def process_game_feed(game_id: str, last_processed_event: Dict[str, Any] = None) -> List[Dict[str, Any]]:
+    """
+    Process the feed for a specific game and extract player stat-change events.
+
+    Args:
+        game_id: The ID of the game to process.
+        last_processed_event: The last event processed for idempotency check.
+
+    Returns:
+        A list of normalized player stat-change events.
+    """
+    feed = _MOCK_FEEDS.get(game_id, [])
+    if not feed:
+        logger.warning(f"No feed found for game_id: {game_id}")
+        return []
+
+    stat_changes = []
+    for event in feed:
+        # Check for idempotency
+        if last_processed_event and is_idempotent(last_processed_event, event):
+            logger.info(f"Skipping duplicate event: {event.get('event_id')}")
+            continue
+
+        # Filter for stat_change events
+        if event.get("type") == "stat_change":
+            normalized_event = {
+                "game_id": game_id,
+                "player_id": event.get("player_id"),
+                "stats": event.get("stats"),
+                "event_id": event.get("event_id"),
+            }
+            stat_changes.append(normalized_event)
+            logger.info(f"Processed stat change for player {normalized_event['player_id']}")
+        else:
+            logger.debug(f"Ignoring non-stat_change event: {event.get('type')}")
+
+    return stat_changes
```
